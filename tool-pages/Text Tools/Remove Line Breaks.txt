<script>
(function(){

var textInput=document.getElementById('textInput'),resultOutput=document.getElementById('resultOutput');
var clearBtn=document.getElementById('clearBtn'),replaceBtn=document.getElementById('replaceBtn');
var copyInputBtn=document.getElementById('copyInputBtn'),copyResultBtn=document.getElementById('copyResultBtn');
var currentMode='single';

document.querySelectorAll('.mode-option').forEach(function(opt){
opt.addEventListener('click',function(){
document.querySelectorAll('.mode-option').forEach(function(o){o.style.borderColor='var(--border)';o.style.background='';o.classList.remove('active')});
opt.style.borderColor='var(--brand)';opt.style.background='var(--brand-soft)';opt.classList.add('active');
currentMode=opt.dataset.mode;
});
});

function process(){
var text=textInput.value.replace(/\r\n?/g,'\n');
var result='';
if(currentMode==='single'){
result=text.replace(/\n/g,' ').replace(/ +/g,' ');
}else if(currentMode==='join'){
result=text.replace(/\n/g,'');
}else{
result=text.replace(/\n\s*\n/g,'\n').replace(/^\n+|\n+$/g,'');
}
resultOutput.value=result;
}

replaceBtn.addEventListener('click',function(){
process();
replaceBtn.innerHTML='<svg fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M5 13l4 4L19 7"/></svg> Done!';
setTimeout(function(){replaceBtn.innerHTML='<svg fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M4 4v5h.582m15.356 2A8.001 8.001 0 004.582 9m0 0H9m11 11v-5h-.581m0 0a8.003 8.003 0 01-15.357-2m15.357 2H15"/></svg> Replace All'},2000);
});

clearBtn.addEventListener('click',function(){textInput.value='';resultOutput.value='';textInput.focus()});

function copyText(el,btn){
if(el.value){
var done=function(){
btn.innerHTML='<svg fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M5 13l4 4L19 7"/></svg> Copied!';
setTimeout(function(){btn.innerHTML='<svg fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M8 16H6a2 2 0 01-2-2V6a2 2 0 012-2h8a2 2 0 012 2v2m-6 12h8a2 2 0 002-2v-8a2 2 0 00-2-2h-8a2 2 0 00-2 2v8a2 2 0 002 2z"/></svg> Copy'},2000);
};
if(navigator.clipboard&&navigator.clipboard.writeText){
navigator.clipboard.writeText(el.value).then(done);
}else{
var ta=document.createElement('textarea');
ta.value=el.value;
ta.style.position='fixed';
ta.style.opacity='0';
document.body.appendChild(ta);
ta.select();
document.execCommand('copy');
document.body.removeChild(ta);
done();
}
}
}

copyInputBtn.addEventListener('click',function(){copyText(textInput,copyInputBtn)});
copyResultBtn.addEventListener('click',function(){copyText(resultOutput,copyResultBtn)});


})();
</script>
