var historyArr=[];var $=function(id){return document.getElementById(id)};
function hexToRgb(h){var r=parseInt(h.slice(1,3),16),g=parseInt(h.slice(3,5),16),b=parseInt(h.slice(5,7),16);return{r:r,g:g,b:b}}
function rgbToHsl(r,g,b){r/=255;g/=255;b/=255;var mx=Math.max(r,g,b),mn=Math.min(r,g,b),d=mx-mn;var h,s,l=(mx+mn)/2;if(!d){h=s=0}else{s=l>.5?d/(2-mx-mn):d/(mx+mn);switch(mx){case r:h=((g-b)/d+(g<b?6:0))/6;break;case g:h=((b-r)/d+2)/6;break;case b:h=((r-g)/d+4)/6;break}h*=360}return{h:Math.round(h),s:Math.round(s*100),l:Math.round(l*100)}}
function rgbToHsv(r,g,b){r/=255;g/=255;b/=255;var mx=Math.max(r,g,b),d=mx-Math.min(r,g,b);var h=0,s=mx?d/mx:0,v=mx;switch(mx){case r:h=((g-b)/d+(g<b?6:0))/6;break;case g:h=((b-r)/d+2)/6;break;case b:h=((r-g)/d+4)/6;break}return{h:Math.round(h*360),s:Math.round(s*100),v:Math.round(v*100)}}
function rotateHue(h,deg){return((h+deg)%360+360)%360}
function hslToHex(h,s,l){s/=100;l/=100;var a=s*Math.min(l,1-l);var f=function(n){var k=(n+h/30)%12;var c=l-a*Math.max(Math.min(k-3,9-k,1),-1);return Math.round(255*c).toString(16).padStart(2,'0')};return'#'+f(0)+f(8)+f(4)}
function update(){
var hex=$('colorPicker').value;var c=hexToRgb(hex);var hsl=rgbToHsl(c.r,c.g,c.b);var hsv=rgbToHsv(c.r,c.g,c.b);
$('preview').style.background=hex;$('hexVal').value=hex.toUpperCase();$('rgbVal').value='rgb('+c.r+', '+c.g+', '+c.b+')';
$('hslVal').value='hsl('+hsl.h+', '+hsl.s+'%, '+hsl.l+'%)';$('rgbaVal').value='rgba('+c.r+', '+c.g+', '+c.b+', 1)';
$('hsvVal').value='hsv('+hsv.h+', '+hsv.s+'%, '+hsv.v+'%)';setHarmonies(hsl.h,hsl.s,hsl.l);addHistory(hex)}
function setHarmonies(h,s,l){
var comp=[hslToHex(rotateHue(h,180),s,l)];var ana=[hslToHex(rotateHue(h,30),s,l),hslToHex(rotateHue(h,-30),s,l)];
var tri=[hslToHex(rotateHue(h,120),s,l),hslToHex(rotateHue(h,240),s,l)];
renderGroup('complementary',comp);renderGroup('analogous',ana);renderGroup('triadic',tri)}
function renderGroup(id,colors){
$(id).innerHTML=colors.map(function(c){return'<div style="width:32px;height:32px;border-radius:6px;border:2px solid var(--border);cursor:pointer;background:'+c+'" title="'+c.toUpperCase()+'" onclick="pickColor(\''+c+'\')"></div>'}).join('')}
function pickColor(hex){$('colorPicker').value=hex;update()}
function addHistory(hex){
historyArr=historyArr.filter(function(c){return c.toLowerCase()!==hex.toLowerCase()});historyArr.unshift(hex);
if(historyArr.length>10)historyArr.pop();
$('historyList').innerHTML=historyArr.map(function(c){return'<div style="width:30px;height:30px;border-radius:6px;border:2px solid var(--surface);cursor:pointer;background:'+c+'" title="'+c.toUpperCase()+'" onclick="pickColor(\''+c+'\')"></div>'}).join('')}
function copy(id,btn){var el=$(id),val=el.value;if(!val)return;var h=btn.innerHTML;btn.innerHTML='&#10003; Copied!';
var done=function(){setTimeout(function(){btn.innerHTML=h},1500)};
if(navigator.clipboard&&navigator.clipboard.writeText){navigator.clipboard.writeText(val).then(done);}
else{var ta=document.createElement('textarea');ta.value=val;ta.style.position='fixed';ta.style.opacity='0';document.body.appendChild(ta);ta.select();try{document.execCommand('copy');done();}catch(e){}document.body.removeChild(ta);}}
function randomColor(){var hex='#'+[0,1,2].map(function(){return Math.floor(Math.random()*256).toString(16).padStart(2,'0')}).join('');$('colorPicker').value=hex;update()}
$('colorPicker').addEventListener('input',update);update();
