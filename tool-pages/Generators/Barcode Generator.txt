<script>
(function(){
  

  var map={code128:'CODE128',code39:'CODE39',ean13:'EAN13',upca:'UPC',itf:'ITF',msi:'MSI',codabar:'codabar'};

  function generate(){
    var val=document.getElementById('barcodeValue').value.trim();
    if(!val)return;
    var type=map[document.getElementById('barcodeType').value]||'CODE128';
    var w=parseInt(document.getElementById('barWidth').value)||2;
    var h=parseInt(document.getElementById('barHeight').value)||100;
    var showTxt=document.getElementById('showText').value==='1';
    var bg=document.getElementById('bgColor').value;
    var lc=document.getElementById('lineColor').value;

    var preview=document.getElementById('preview');
    preview.innerHTML='<svg id="barcodeSvg"></svg>';
    try{
      JsBarcode('#barcodeSvg',val,{format:type,width:w,height:h,displayValue:showTxt,background:bg,lineColor:lc,font:'monospace',fontSize:14,margin:10});
      document.getElementById('dlWrap').style.display='block';
    }catch(e){preview.innerHTML='<p style="color:#DC2626;font-weight:600">Error: '+e.message+'</p>';document.getElementById('dlWrap').style.display='none';}
  }

  document.getElementById('generateBtn').addEventListener('click',generate);

  document.getElementById('downloadPng').addEventListener('click',function(){
    var svg=document.getElementById('barcodeSvg');if(!svg)return;
    var c=document.createElement('canvas');var ctx=c.getContext('2d');
    var s=new XMLSerializer().serializeToString(svg);
    var img=new Image();img.onload=function(){c.width=img.width*2;c.height=img.height*2;ctx.fillStyle='#fff';ctx.fillRect(0,0,c.width,c.height);ctx.drawImage(img,0,0,c.width,c.height);
    var a=document.createElement('a');a.href=c.toDataURL('image/png');a.download='barcode.png';a.click();};
    img.src='data:image/svg+xml;base64,'+btoa(unescape(encodeURIComponent(s)));
  });

  document.getElementById('downloadSvg').addEventListener('click',function(){
    var svg=document.getElementById('barcodeSvg');if(!svg)return;
    var s=new XMLSerializer().serializeToString(svg);
    var blob=new Blob([s],{type:'image/svg+xml'});var url=URL.createObjectURL(blob);
    var a=document.createElement('a');a.href=url;a.download='barcode.svg';a.click();URL.revokeObjectURL(url);
  });

  generate();
})();
</script>
