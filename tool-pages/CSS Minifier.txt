<script>
(function(){
  

  function minifyCSS(css){
    css=css.replace(/\/\*[\s\S]*?\*\//g,'');
    var strings=[],placeholders=[];
    css=css.replace(/("(?:[^"\\]|\\.)*"|'(?:[^'\\]|\\.)*')/g,function(m){strings.push(m);placeholders.push('__CSSSTR'+strings.length+'__');return placeholders[placeholders.length-1];});
    css=css.replace(/\s*([{}:;,])\s*/g,'$1').replace(/;}/g,'}').replace(/\s+/g,' ');
    for(var i=0;i<strings.length;i++)css=css.replace(placeholders[i],function(){return strings[i];});
    return css.trim();
  }
  function beautifyCSS(css){
    css=css.replace(/\/\*[\s\S]*?\*\//g,'');
    var strings=[],placeholders=[];
    css=css.replace(/("(?:[^"\\]|\\.)*"|'(?:[^'\\]|\\.)*')/g,function(m){strings.push(m);placeholders.push('__CSSSTR'+strings.length+'__');return placeholders[placeholders.length-1];});
    css=css.replace(/\s+/g,' ').trim();
    var result='',indent=0;
    css.split('').forEach(function(c){
      if(c==='{'){result+=' {\n';indent++;result+=Array(indent*2+1).join(' ');}
      else if(c==='}'){result+='\n';indent--;result+=Array(indent*2+1).join(' ');result+='}\n';result+=Array(indent*2+1).join(' ');}
      else if(c===';'){result+=';\n';result+=Array(indent*2+1).join(' ');}
      else{result+=c;}
    });
    result=result.replace(/\n\s*\n/g,'\n').trim();
    for(var i=0;i<strings.length;i++)result=result.replace(placeholders[i],function(){return strings[i];});
    return result;
  }
  function updateStats(orig,min){
    var o=orig.length,m=min.length,pct=o>0?((1-m/o)*100).toFixed(1):0;
    document.getElementById('statsDisplay').innerHTML='<div class="stats-row"><div class="stat-box"><div class="stat-num">'+o+'</div><div class="stat-label">Original</div></div><div class="stat-box"><div class="stat-num">'+m+'</div><div class="stat-label">Minified</div></div><div class="stat-box"><div class="stat-num">'+pct+'%</div><div class="stat-label">Saved</div></div></div>';
  }
  document.getElementById('minifyBtn').addEventListener('click',function(){
    var css=document.getElementById('inputCSS').value;if(!css){alert('Please enter CSS code.');return;}
    var result=minifyCSS(css);updateStats(css,result);document.getElementById('output').textContent=result;
  });
  document.getElementById('beautifyBtn').addEventListener('click',function(){
    var css=document.getElementById('inputCSS').value;if(!css){alert('Please enter CSS code.');return;}
    var result=beautifyCSS(css);document.getElementById('output').textContent=result;document.getElementById('statsDisplay').innerHTML='';
  });
  document.getElementById('copyBtn').addEventListener('click',function(){
    var btn=document.getElementById('copyBtn'),t=document.getElementById('output').textContent;if(!t)return;
    var h=btn.innerHTML;
    var done=function(){btn.innerHTML='<svg fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M9 12l2 2 4-4"/></svg> Copied!';setTimeout(function(){btn.innerHTML=h;},2000);};
    if(navigator.clipboard&&navigator.clipboard.writeText){navigator.clipboard.writeText(t).then(done);}
    else{var ta=document.createElement('textarea');ta.value=t;ta.style.position='fixed';ta.style.opacity='0';document.body.appendChild(ta);ta.select();try{document.execCommand('copy');done();}catch(e){}document.body.removeChild(ta);}
  });
})();
</script>
