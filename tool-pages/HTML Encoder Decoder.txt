<script>
(function(){
  
  var enc={'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'};
  var dec={'&amp;':'&','&lt;':'<','&gt;':'>','&quot;':'"', '&#39;':"'",'&#x27;':"'"};
  document.getElementById('encodeBtn').addEventListener('click',function(){
    var t=document.getElementById('inputText').value;if(!t){alert('Please enter text.');return;}
    var r=t.replace(/[&<>"']/g,function(m){return enc[m];});
    document.getElementById('output').textContent=r;
  });
  document.getElementById('decodeBtn').addEventListener('click',function(){
    var t=document.getElementById('inputText').value;if(!t){alert('Please enter text.');return;}
    var r=t.replace(/&amp;|&lt;|&gt;|&quot;|&#39;|&#x27;/g,function(m){return dec[m];});
    document.getElementById('output').textContent=r;
  });
  document.getElementById('copyBtn').addEventListener('click',function(){
    var btn=document.getElementById('copyBtn'),t=document.getElementById('output').textContent;if(!t)return;
    var h=btn.innerHTML;
    var done=function(){btn.innerHTML='<svg fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M9 12l2 2 4-4"/></svg> Copied!';setTimeout(function(){btn.innerHTML=h;},2000);};
    if(navigator.clipboard&&navigator.clipboard.writeText){navigator.clipboard.writeText(t).then(done);}
    else{var ta=document.createElement('textarea');ta.value=t;ta.style.position='fixed';ta.style.opacity='0';document.body.appendChild(ta);ta.select();try{document.execCommand('copy');done();}catch(e){}document.body.removeChild(ta);}
  });
})();
</script>
