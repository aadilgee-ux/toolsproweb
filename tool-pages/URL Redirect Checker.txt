<script>
var $=function(s){return document.querySelector(s)};
var statusClass=function(c){return c>=200&&c<300?'status-2xx':c>=300&&c<400?'status-3xx':c>=400&&c<500?'status-4xx':'status-5xx'};
var STATUS_TEXT={200:'OK',201:'Created',204:'No Content',301:'Moved Permanently',302:'Found',303:'See Other',304:'Not Modified',307:'Temporary Redirect',308:'Permanent Redirect',400:'Bad Request',401:'Unauthorized',403:'Forbidden',404:'Not Found',410:'Gone',429:'Too Many Requests',500:'Internal Server Error',502:'Bad Gateway',503:'Service Unavailable',504:'Gateway Timeout'};
var escapeHtml=function(s){return s.replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;')};
async function traceRedirect(url){
var chain=[],current=url,maxHops=10,visited=new Set();visited.add(url);
for(var i=0;i<maxHops;i++){
try{
var proxy='https://api.allorigins.win/get?url='+encodeURIComponent(current);
var r=await fetch(proxy),j=await r.json();
var code=(j&&j.status&&j.status.http_code)?j.status.http_code:0;
var finalUrl=(j&&j.status&&j.status.url)?j.status.url:current;
chain.push({url:current,status:code,statusText:STATUS_TEXT[code]||('Status '+code)});
if(code<300||code===0||!j.status){break;}
if(visited.has(finalUrl))break;
visited.add(finalUrl);current=finalUrl;
}catch(e){chain.push({url:current,status:0,statusText:'Failed to fetch'});break;}
}
return chain;}
function renderChain(chain,idx){
if(chain.length===0)return'';var html='<div style="margin-bottom:24px"><h3 style="font-size:.9rem;margin-bottom:10px;color:var(--text-secondary)">Result '+(idx+1)+'</h3><div style="display:flex;flex-direction:column;gap:0">';
chain.forEach(function(hop,i){var sClass=statusClass(hop.status);var isFinal=i===chain.length-1;
var label=i===0?'Original URL':isFinal?'Final Destination':'Hop '+i;
html+='<div class="chain-item"><div style="min-width:80px"><div style="font-size:.7rem;color:var(--text-secondary);text-transform:uppercase;letter-spacing:.5px;margin-bottom:2px">'+label+'</div><span class="status-badge '+sClass+'">'+(hop.status||'ERR')+'</span></div><div><div style="font-size:.7rem;color:var(--text-secondary);text-transform:uppercase;letter-spacing:.5px;margin-bottom:2px">URL</div><div style="font-size:.85rem;word-break:break-all;color:var(--text-secondary)">'+escapeHtml(hop.url)+'</div></div></div>';
if(!isFinal)html+='<div class="chain-arrow">&#8595;</div>';});
html+='</div></div>';return html;}
function renderStats(chain){
var redirects=chain.length>1?chain.length-1:0;var finalUrl=chain.length>0?chain[chain.length-1].url:'N/A';
return '<div style="display:flex;gap:16px;flex-wrap:wrap;margin-bottom:20px"><div class="stat-box"><div class="stat-val">'+redirects+'</div><div class="stat-lbl">Total Redirects</div></div><div class="stat-box" style="flex:3"><div class="stat-val" style="font-size:.9rem;word-break:break-all">'+escapeHtml(finalUrl)+'</div><div class="stat-lbl">Final Destination</div></div></div>';}
async function checkUrl(url){if(!url)throw new Error('Please enter a URL');try{new URL(url);}catch(e){throw new Error('Invalid URL format');}return await traceRedirect(url);}
document.getElementById('checkBtn').onclick=async function(){
var url=document.getElementById('urlInput').value.trim();
var err=document.getElementById('errorMsg'),res=document.getElementById('results'),ld=document.getElementById('loading');
err.style.display='none';res.style.display='none';
if(!url){err.textContent='Please enter a URL';err.style.display='block';return;}
if(!url.startsWith('http')){err.textContent='URL must start with http:// or https://';err.style.display='block';return;}
ld.classList.add('show');this.disabled=true;
try{var chain=await checkUrl(url);res.innerHTML=renderStats(chain)+renderChain(chain,0);res.style.display='block';}
catch(e){err.textContent=e.message;err.style.display='block';}
ld.classList.remove('show');document.getElementById('checkBtn').disabled=false;};
document.getElementById('bulkCheckBtn').onclick=async function(){
var lines=document.getElementById('bulkInput').value.split('\n').map(function(l){return l.trim()}).filter(Boolean);
var err=document.getElementById('errorMsg'),res=document.getElementById('results'),ld=document.getElementById('loading');
err.style.display='none';res.style.display='none';
if(!lines.length){err.textContent='Enter at least one URL';err.style.display='block';return;}
ld.classList.add('show');this.disabled=true;var allHtml='';
for(var i=0;i<lines.length;i++){try{
if(!lines[i].startsWith('http://')&&!lines[i].startsWith('https://')){allHtml+='<div style="margin-bottom:24px"><h3 style="font-size:.9rem;margin-bottom:10px;color:var(--text-secondary)">Result '+(i+1)+'</h3><div style="padding:16px;background:var(--bg);border-radius:8px;color:#EF4444">Invalid URL: '+escapeHtml(lines[i])+'</div></div>';continue;}
var chain=await checkUrl(lines[i]);allHtml+=renderStats(chain)+renderChain(chain,i);
}catch(e){allHtml+='<div style="margin-bottom:24px"><h3 style="font-size:.9rem;margin-bottom:10px;color:var(--text-secondary)">Result '+(i+1)+'</h3><div style="padding:16px;background:var(--bg);border-radius:8px;color:#EF4444">'+escapeHtml(e.message)+'</div></div>';}
}res.innerHTML=allHtml;res.style.display='block';ld.classList.remove('show');document.getElementById('bulkCheckBtn').disabled=false;};
document.getElementById('urlInput').addEventListener('keydown',function(e){if(e.key==='Enter')document.getElementById('checkBtn').click();});
</script>
