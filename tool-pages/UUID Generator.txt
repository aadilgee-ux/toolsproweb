<script>
(function(){
  

  function uuidV4(){return'xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx'.replace(/[xy]/g,function(c){var x=Math.random()*16|0,y=c==='x'?x:(x&0x3|0x8);return y.toString(16);});}
  function uuidV1(){var t=Date.now(),th=(t/0x100000000)>>>0,tl=(t&0xFFFFFFFF)>>>0;var cs=Math.random()*0xFFFF|0;var n=Math.random()*0xFFFF|0|0x1000;return hex32(tl)+'-'+hex16(th&0xFFFF)+'-'+hex16(0x1000|((n>>12)&0xFFF))+'-'+hex16(0x8000|(n&0x3FFF))+'-'+hex16(cs)+hex16(Math.random()*0xFFFF|0);}
  function hex16(h){return('0000'+h.toString(16)).slice(-4);}
  function hex32(h){return('00000000'+h.toString(16)).slice(-8);}

  function formatUUID(u,f){
    var d=u;
    if(f==='no-dash')d=d.replace(/-/g,'');
    else if(f==='upper')d=d.toUpperCase();
    else if(f==='braces')d='{'+d+'}';
    return d;
  }

  var uuids=[];

  function render(){
    var list=document.getElementById('uuidList');list.innerHTML='';
    var prefix=document.getElementById('uuidPrefix').value||'';
    var fmt=document.getElementById('uuidFormat').value;
    uuids.forEach(function(u,i){
      var display=prefix+formatUUID(u,fmt);
      var div=document.createElement('div');div.className='uuid-item';
      div.innerHTML='<span class="uuid-text">'+display+'</span><button title="Copy" data-idx="'+i+'">&#128203;</button>';
      list.appendChild(div);
    });
    list.querySelectorAll('button[data-idx]').forEach(function(b){b.addEventListener('click',function(){
      var idx=+b.dataset.idx;var u=uuids[idx];
      var display=(prefix||'')+formatUUID(u,fmt);
      copyText(display);
      b.textContent='&#10003;';setTimeout(function(){b.innerHTML='&#128203;';},1000);
    });});
  }

  function copyText(text){
    if(navigator.clipboard&&navigator.clipboard.writeText){navigator.clipboard.writeText(text);return;}
    var ta=document.createElement('textarea');ta.value=text;ta.style.position='fixed';ta.style.left='-9999px';
    document.body.appendChild(ta);ta.select();document.execCommand('copy');document.body.removeChild(ta);
  }

  document.getElementById('generateBtn').addEventListener('click',function(){
    var count=Math.min(parseInt(document.getElementById('uuidCount').value)||10,100);
    var ver=document.getElementById('uuidVersion').value;
    uuids=[];
    for(var i=0;i<count;i++){uuids.push(ver==='1'?uuidV1():uuidV4());}
    render();
  });

  document.getElementById('copyAll').addEventListener('click',function(){
    var prefix=document.getElementById('uuidPrefix').value||'';
    var fmt=document.getElementById('uuidFormat').value;
    var text=uuids.map(function(u){return prefix+formatUUID(u,fmt);}).join('\n');
    copyText(text);
    document.getElementById('copyAll').textContent='Copied!';setTimeout(function(){document.getElementById('copyAll').innerHTML='&#128203; Copy All';},1500);
  });

  document.getElementById('clearBtn').addEventListener('click',function(){uuids=[];document.getElementById('uuidList').innerHTML='';});
  document.getElementById('uuidFormat').addEventListener('change',render);
  document.getElementById('uuidPrefix').addEventListener('input',render);

  document.getElementById('generateBtn').click();
})();
</script>
