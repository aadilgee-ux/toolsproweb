<script>
(function() {

  var pathsContainer = document.getElementById('pathsContainer');
  var addPathBtn = document.getElementById('addPathBtn');
  var generateBtn = document.getElementById('generateBtn');
  var robotsOutput = document.getElementById('robotsOutput');
  var copyBtn = document.getElementById('copyBtn');
  var downloadBtn = document.getElementById('downloadBtn');

  addPathBtn.addEventListener('click', function() {
    var row = document.createElement('div');
    row.className = 'path-row';
    row.innerHTML = '<select class="form-select"><option value="Disallow">Disallow</option><option value="Allow">Allow</option></select><input class="form-input" placeholder="/path/to/folder" type="text"/><button class="path-remove" type="button"><svg fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M19 7l-.867 12.142A2 2 0 0116.138 21H7.862a2 2 0 01-1.995-1.858L5 7m5 4v6m4-6v6m1-10V4a1 1 0 00-1-1h-4a1 1 0 00-1 1v3M4 7h16"/></svg></button>';
    row.querySelector('.path-remove').addEventListener('click', function() { row.remove(); });
    pathsContainer.appendChild(row);
  });

  pathsContainer.querySelector('.path-remove').addEventListener('click', function() { this.closest('.path-row').remove(); });

  generateBtn.addEventListener('click', function() {
    var ua = document.getElementById('userAgent').value;
    var sitemap = document.getElementById('sitemapUrl').value.trim();
    var lines = ['# Robots.txt generated by Free Online Tools Corner', '# Visit us at https://example.com', '', 'User-agent: ' + ua];

    var rows = pathsContainer.querySelectorAll('.path-row');
    rows.forEach(function(row) {
      var action = row.querySelector('.form-select').value;
      var path = row.querySelector('.form-input').value.trim();
      if (path) lines.push(action + ': ' + path);
    });

    if (sitemap) lines.push('', 'Sitemap: ' + sitemap);

    robotsOutput.textContent = lines.join('\n');
  });

  copyBtn.addEventListener('click', function() {
    if (robotsOutput.textContent) {
      var done = function() {
        copyBtn.innerHTML = '<svg fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M5 13l4 4L19 7"/></svg> Copied!';
        setTimeout(function() { copyBtn.innerHTML = '<svg fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M8 16H6a2 2 0 01-2-2V6a2 2 0 012-2h8a2 2 0 012 2v2m-6 12h8a2 2 0 002-2v-8a2 2 0 00-2-2h-8a2 2 0 00-2 2v8a2 2 0 002 2z"/></svg> Copy'; }, 2000);
      };
      if (navigator.clipboard && navigator.clipboard.writeText) {
        navigator.clipboard.writeText(robotsOutput.textContent).then(done);
      } else {
        var ta = document.createElement('textarea');
        ta.value = robotsOutput.textContent;
        ta.style.position = 'fixed';
        ta.style.opacity = '0';
        document.body.appendChild(ta);
        ta.select();
        document.execCommand('copy');
        document.body.removeChild(ta);
        done();
      }
    }
  });

  downloadBtn.addEventListener('click', function() {
    if (robotsOutput.textContent) {
      var blob = new Blob([robotsOutput.textContent], { type: 'text/plain' });
      var a = document.createElement('a');
      a.href = URL.createObjectURL(blob);
      a.download = 'robots.txt';
      a.click();
      URL.revokeObjectURL(a.href);
    }
  });

  
})();
</script>
