<script>
(function(){
  

  function sanitizeUrl(u){
    u=u.trim().replace(/^[\s\uFEFF\xA0]+|[\s\uFEFF\xA0]+$/g,'');
    if(!u)return '';
    var bad=/^(javascript|data|vbscript):/i;
    if(bad.test(u))return '#';
    return u.replace(/"/g,'&quot;');
  }
  function escapeHtml(s){
    return String(s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');
  }
  function md2html(md){
    var html=escapeHtml(md);
    html=html.replace(/```(\w*)\n([\s\S]*?)```/g,function(m,lang,code){return '<pre><code class="language-'+lang+'">'+code+'</code></pre>';});
    html=html.replace(/`([^`]+)`/g,'<code>$1</code>');
    html=html.replace(/^######\s+(.+)$/gm,'<h6>$1</h6>');
    html=html.replace(/^#####\s+(.+)$/gm,'<h5>$1</h5>');
    html=html.replace(/^####\s+(.+)$/gm,'<h4>$1</h4>');
    html=html.replace(/^###\s+(.+)$/gm,'<h3>$1</h3>');
    html=html.replace(/^##\s+(.+)$/gm,'<h2>$1</h2>');
    html=html.replace(/^#\s+(.+)$/gm,'<h1>$1</h1>');
    html=html.replace(/^---$/gm,'<hr>');
    html=html.replace(/^>\s+(.+)$/gm,'<blockquote>$1</blockquote>');
    html=html.replace(/\*\*(.+?)\*\*/g,'<strong>$1</strong>');
    html=html.replace(/\*(.+?)\*/g,'<em>$1</em>');
    html=html.replace(/~~(.+?)~~/g,'<del>$1</del>');
    html=html.replace(/!\[([^\]]*)\]\(([^)]+)\)/g,function(m,a,u){return '<img alt="'+a.replace(/"/g,'&quot;')+'" src="'+sanitizeUrl(u)+'"/>';});
    html=html.replace(/\[([^\]]+)\]\(([^)]+)\)/g,function(m,t,u){return '<a href="'+sanitizeUrl(u)+'">'+t+'</a>';});
    html=html.replace(/^\s*[-*+]\s+(.+)$/gm,'<li>$1</li>');
    html=html.replace(/^\s*\d+\.\s+(.+)$/gm,'<li>$1</li>');
    html=html.replace(/(<li>.*<\/li>\n?)+/g,function(m){return '<ul>'+m+'</ul>';});
    html=html.replace(/\n\n/g,'</p><p>');
    html='<p>'+html+'</p>';
    html=html.replace(/<p>\s*<(h[1-6]|ul|ol|blockquote|pre|hr)/g,'<$1');
    html=html.replace(/<\/(h[1-6]|ul|ol|blockquote|pre)>\s*<\/p>/g,'</$1>');
    html=html.replace(/<p>\s*<\/p>/g,'');
    html=html.replace(/<p><hr><\/p>/g,'<hr>');
    return html;
  }

  document.getElementById('convertBtn').addEventListener('click',function(){
    var md=document.getElementById('inputMD').value;if(!md){alert('Please enter Markdown text.');return;}
    var html=md2html(md);
    document.getElementById('htmlOutput').textContent=html;
    document.getElementById('preview').innerHTML=html;
  });

  document.getElementById('copyBtn').addEventListener('click',function(){
    var btn=document.getElementById('copyBtn'),t=document.getElementById('htmlOutput').textContent;if(!t)return;
    var h=btn.innerHTML;
    var done=function(){btn.innerHTML='<svg fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M9 12l2 2 4-4"/></svg> Copied!';setTimeout(function(){btn.innerHTML=h;},2000);};
    if(navigator.clipboard&&navigator.clipboard.writeText){navigator.clipboard.writeText(t).then(done);}
    else{var ta=document.createElement('textarea');ta.value=t;ta.style.position='fixed';ta.style.opacity='0';document.body.appendChild(ta);ta.select();try{document.execCommand('copy');done();}catch(e){}document.body.removeChild(ta);}
  });
})();
</script>
