<script>
(function() {

  var jsonInput = document.getElementById('jsonInput');
  var jsonOutput = document.getElementById('jsonOutput');
  var formatBtn = document.getElementById('formatBtn');
  var minifyBtn = document.getElementById('minifyBtn');
  var compressBtn = document.getElementById('compressBtn');
  var copyBtn = document.getElementById('copyBtn');
  var clearBtn = document.getElementById('clearBtn');

  function formatJSON(text, indent) {
    try {
      var parsed = JSON.parse(text);
      return { success: true, result: JSON.stringify(parsed, null, indent || 2) };
    } catch (e) {
      return { success: false, error: e.message };
    }
  }

  function processJSON(indent) {
    var text = jsonInput.value.trim();
    if (!text) { jsonOutput.innerHTML = ''; return; }
    var res = formatJSON(text, indent);
    if (res.success) {
      jsonOutput.className = 'result-output';
      jsonOutput.textContent = res.result;
    } else {
      jsonOutput.className = 'result-output result-error';
      jsonOutput.textContent = 'Error: ' + (res.error || 'Invalid JSON');
    }
  }

  formatBtn.addEventListener('click', function() { processJSON(2); });
  minifyBtn.addEventListener('click', function() { processJSON(0); });
  compressBtn.addEventListener('click', function() { processJSON(0); });

  clearBtn.addEventListener('click', function() { jsonInput.value = ''; jsonOutput.innerHTML = ''; jsonInput.focus(); });

  copyBtn.addEventListener('click', function() {
    var text = jsonOutput.textContent;
    if (!text) return;
    var h = copyBtn.innerHTML;
    var done = function() {
      copyBtn.innerHTML = '<svg fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M5 13l4 4L19 7"/></svg> Copied!';
      setTimeout(function() { copyBtn.innerHTML = h; }, 2000);
    };
    if (navigator.clipboard && navigator.clipboard.writeText) {
      navigator.clipboard.writeText(text).then(done);
    } else {
      var ta = document.createElement('textarea');
      ta.value = text;
      ta.style.position = 'fixed';
      ta.style.opacity = '0';
      document.body.appendChild(ta);
      ta.select();
      try { document.execCommand('copy'); done(); } catch (e) {}
      document.body.removeChild(ta);
    }
  });

  
})();
</script>
