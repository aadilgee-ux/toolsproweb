<script>
(function() {

  /* ===== PASSWORD GENERATOR ===== */
  var pwOutput = document.getElementById('pwOutput');
  var pwLength = document.getElementById('pwLength');
  var pwLengthVal = document.getElementById('pwLengthVal');
  var chkUpper = document.getElementById('chkUpper');
  var chkLower = document.getElementById('chkLower');
  var chkNum = document.getElementById('chkNum');
  var chkSym = document.getElementById('chkSym');
  var chkUpperWrap = document.getElementById('chkUpperWrap');
  var chkLowerWrap = document.getElementById('chkLowerWrap');
  var chkNumWrap = document.getElementById('chkNumWrap');
  var chkSymWrap = document.getElementById('chkSymWrap');
  var generateBtn = document.getElementById('generateBtn');
  var copyPwBtn = document.getElementById('copyPwBtn');
  var showPwBtn = document.getElementById('showPwBtn');
  var resetBtn = document.getElementById('resetBtn');
  var strengthBar = document.getElementById('strengthBar');
  var strengthLabel = document.getElementById('strengthLabel');
  var strengthScore = document.getElementById('strengthScore');
  var toastEl = document.getElementById('toast');
  var eyeOpen = document.getElementById('eyeOpen');
  var eyeClosed = document.getElementById('eyeClosed');

  var CHARS_UPPER = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ';
  var CHARS_LOWER = 'abcdefghijklmnopqrstuvwxyz';
  var CHARS_NUMS = '0123456789';
  var CHARS_SYM = '!@#$%^&*()_+-=[]{}|;:,.<>?';
  var passwordVisible = false;
  var currentPassword = '';

  function generatePassword() {
    var length = parseInt(pwLength.value, 10);
    var chars = '';
    var guaranteed = [];

    if (chkUpper.checked) { chars += CHARS_UPPER; guaranteed.push(CHARS_UPPER); }
    if (chkLower.checked) { chars += CHARS_LOWER; guaranteed.push(CHARS_LOWER); }
    if (chkNum.checked)   { chars += CHARS_NUMS; guaranteed.push(CHARS_NUMS); }
    if (chkSym.checked)   { chars += CHARS_SYM; guaranteed.push(CHARS_SYM); }

    if (chars.length === 0) {
      pwOutput.textContent = 'Select at least one option';
      pwOutput.style.color = 'var(--text-secondary)';
      currentPassword = '';
      updateStrength('');
      return;
    }

    var array = new Uint32Array(length);
    window.crypto.getRandomValues(array);

    var pw = [];
    for (var i = 0; i < length; i++) {
      pw.push(chars[array[i] % chars.length]);
    }

    for (var g = 0; g < guaranteed.length && g < length; g++) {
      var rndIdx = new Uint32Array(1);
      window.crypto.getRandomValues(rndIdx);
      var pos = rndIdx[0] % length;
      var charIdx = new Uint32Array(1);
      window.crypto.getRandomValues(charIdx);
      pw[pos] = guaranteed[g][charIdx[0] % guaranteed[g].length];
    }

    currentPassword = pw.join('');
    pwOutput.textContent = passwordVisible ? currentPassword : maskPassword(currentPassword);
    pwOutput.style.color = '';
    updateStrength(currentPassword);
  }

  function maskPassword(pw) {
    return pw.replace(/./g, function() { return '\u2022'; });
  }

  function calcStrength(pw) {
    if (!pw) return { score: 0, label: '—', color: 'var(--text-secondary)', width: '0%' };
    var score = 0;
    var len = pw.length;
    if (len >= 8) score += 1;
    if (len >= 12) score += 1;
    if (len >= 16) score += 1;
    if (len >= 24) score += 1;
    var hasUpper = /[A-Z]/.test(pw);
    var hasLower = /[a-z]/.test(pw);
    var hasNum = /[0-9]/.test(pw);
    var hasSym = /[^A-Za-z0-9]/.test(pw);
    var types = (hasUpper ? 1 : 0) + (hasLower ? 1 : 0) + (hasNum ? 1 : 0) + (hasSym ? 1 : 0);
    score += types;
    if (len >= 16 && types >= 3) score += 1;

    var label, color, width;
    if (score <= 3) { label = 'Weak'; color = '#DC2626'; width = '25%'; }
    else if (score <= 5) { label = 'Medium'; color = '#F59E0B'; width = '50%'; }
    else if (score <= 7) { label = 'Strong'; color = '#22C55E'; width = '75%'; }
    else { label = 'Very Strong'; color = '#16A34A'; width = '100%'; }

    return { score: score, label: label, color: color, width: width };
  }

  function updateStrength(pw) {
    var s = calcStrength(pw);
    strengthBar.style.width = s.width;
    strengthBar.style.background = s.color;
    strengthLabel.textContent = s.label;
    strengthLabel.style.color = s.color;
    strengthScore.textContent = pw ? 'Score: ' + s.score + '/9' : '';
  }

  function showToast(msg) {
    if (!toastEl) return;
    toastEl.textContent = msg;
    toastEl.classList.add('show');
    setTimeout(function() { toastEl.classList.remove('show'); }, 2200);
  }

  function syncCheckboxes() {
    var items = [
      { el: chkUpper, wrap: chkUpperWrap },
      { el: chkLower, wrap: chkLowerWrap },
      { el: chkNum, wrap: chkNumWrap },
      { el: chkSym, wrap: chkSymWrap }
    ];
    items.forEach(function(item) {
      if (item.el.checked) item.wrap.classList.add('checked');
      else item.wrap.classList.remove('checked');
    });
  }

  function ensureAtLeastOne() {
    if (!chkUpper.checked && !chkLower.checked && !chkNum.checked && !chkSym.checked) {
      chkLower.checked = true;
      chkLowerWrap.classList.add('checked');
    }
  }

  pwLength.addEventListener('input', function() {
    pwLengthVal.textContent = pwLength.value;
    generatePassword();
  });

  [chkUpper, chkLower, chkNum, chkSym].forEach(function(cb) {
    cb.addEventListener('change', function() {
      syncCheckboxes();
      ensureAtLeastOne();
      generatePassword();
    });
  });

  generateBtn.addEventListener('click', function() {
    generatePassword();
    showToast('New password generated!');
  });

  copyPwBtn.addEventListener('click', function() {
    if (!currentPassword) return;
    if (navigator.clipboard && navigator.clipboard.writeText) {
      navigator.clipboard.writeText(currentPassword).then(function() {
        showToast('Password copied to clipboard!');
      });
    } else {
      var ta = document.createElement('textarea');
      ta.value = currentPassword;
      ta.style.position = 'fixed';
      ta.style.left = '-9999px';
      document.body.appendChild(ta);
      ta.select();
      document.execCommand('copy');
      document.body.removeChild(ta);
      showToast('Password copied to clipboard!');
    }
  });

  showPwBtn.addEventListener('click', function() {
    passwordVisible = !passwordVisible;
    pwOutput.textContent = passwordVisible ? currentPassword : maskPassword(currentPassword);
    eyeOpen.style.display = passwordVisible ? 'block' : 'none';
    eyeClosed.style.display = passwordVisible ? 'none' : 'block';
    showPwBtn.childNodes[showPwBtn.childNodes.length - 1].textContent = passwordVisible ? ' Hide' : ' Show';
  });

  resetBtn.addEventListener('click', function() {
    pwLength.value = 16;
    pwLengthVal.textContent = '16';
    chkUpper.checked = true;
    chkLower.checked = true;
    chkNum.checked = true;
    chkSym.checked = true;
    passwordVisible = false;
    eyeOpen.style.display = 'none';
    eyeClosed.style.display = 'block';
    syncCheckboxes();
    generatePassword();
    showToast('Options reset to default!');
  });

  /* ===== YEAR ===== */
  

  /* ===== INIT ===== */
  generatePassword();
})();
</script>
