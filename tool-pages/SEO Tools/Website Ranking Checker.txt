<script>
(function(){
var inText=document.getElementById('inText'),outText=document.getElementById('outText');
var runBtn=document.getElementById('runBtn'),clearBtn=document.getElementById('clearBtn'),copyBtn=document.getElementById('copyBtn'),downloadBtn=document.getElementById('downloadBtn');
var addBtn=document.getElementById('addBtn');

function copyText(text,btn,done,orig){
 function d(){btn.textContent=done;setTimeout(function(){btn.textContent=orig;},2000);}
 if(navigator.clipboard&&navigator.clipboard.writeText){navigator.clipboard.writeText(text).then(d);}
 else{var ta=document.createElement('textarea');ta.value=text;ta.style.position='fixed';ta.style.opacity='0';document.body.appendChild(ta);ta.select();document.execCommand('copy');document.body.removeChild(ta);d();}
}
function escHtml(s){return String(s==null?'':s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;');}
var kwEl=document.getElementById('kw'),posEl=document.getElementById('pos'),engineSel=document.getElementById('engine'),rdateEl=document.getElementById('rdate');

var STORE_KEY='tpw_rankings';
function loadAll(){try{return JSON.parse(localStorage.getItem(STORE_KEY))||[];}catch(e){return [];}}
function saveAll(a){localStorage.setItem(STORE_KEY,JSON.stringify(a));}
function render(){
 var a=loadAll();var wrap=document.getElementById('tableWrap');
 if(!a.length){wrap.innerHTML='<p class="empty-state">No entries yet. Add your first keyword position above.</p>';return;}
 var rows='';
 var byKw={};a.forEach(function(e,i){var k=e.kw.toLowerCase();if(!byKw[k])byKw[k]=[];byKw[k].push(i);});
 a.slice().reverse().forEach(function(e,i){
  var prev=null;var list=byKw[e.kw.toLowerCase()]||[];var idx=list.indexOf(a.indexOf(e));if(idx>0)prev=a[list[idx-1]];
  var trend='';
  if(prev&&prev.pos!==undefined&&e.pos!==undefined){if(e.pos<prev.pos)trend='<span class="trend up">▲ +'+(prev.pos-e.pos)+'</span>';else if(e.pos>prev.pos)trend='<span class="trend down">▼ -'+(e.pos-prev.pos)+'</span>';else trend='<span class="trend flat">=</span>';}
  rows+='<tr><td>'+escHtml(e.kw)+'</td><td>'+escHtml(e.engine||'')+'</td><td>'+escHtml(String(e.pos))+' '+trend+'</td><td>'+escHtml(e.date||'-')+'</td><td><button class="row-del" data-i="'+i+'" type="button">Delete</button></td></tr>';
 });
 wrap.innerHTML='<table class="rank-table"><thead><tr><th>Keyword</th><th>Engine</th><th>Position</th><th>Date</th><th></th></tr></thead><tbody>'+rows+'</tbody></table>';
 wrap.querySelectorAll('.row-del').forEach(function(b){b.addEventListener('click',function(){var arr=loadAll();arr.splice(b.getAttribute('data-i'),1);saveAll(arr);render();});});
}
function escHtml(s){return String(s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;');}
addBtn.addEventListener('click',function(){
 var kw=kwEl.value.trim();if(!kw){alert('Please enter a keyword.');return;}
 var pos=parseInt(posEl.value,10);if(!pos||pos<1){alert('Please enter a valid position (1 or higher).');return;}
 var arr=loadAll();arr.push({kw:kw,pos:pos,engine:engineSel.value,date:rdateEl.value||new Date().toISOString().slice(0,10)});saveAll(arr);
 kwEl.value='';posEl.value='1';render();kwEl.focus();
});
clearBtn.addEventListener('click',function(){if(confirm('Delete all ranking entries?')){localStorage.removeItem(STORE_KEY);render();}});
copyBtn.addEventListener('click',function(){
 var a=loadAll();if(!a.length)return;
 var csv='Keyword,Engine,Position,Date\n'+a.map(function(e){return '"'+String(e.kw).replace(/"/g,'""')+'","'+e.engine+'","'+e.pos+'","'+e.date+'"';}).join('\n');
 copyText(csv,this,'Copied!','Copy Table (CSV)');
});
render();
function runSafe(){try{run();}catch(e){if(outText)outText.value='Error: '+e.message;else if(document.getElementById('outBox'))document.getElementById('outBox').innerHTML='<p class="muted">Error: '+String(e.message).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;')+'</p>';}}
if(runBtn)runBtn.addEventListener('click',runSafe);
if(clearBtn)clearBtn.addEventListener('click',function(){if(inText)inText.value='';if(outText)outText.value='';var ob=document.getElementById('outBox');if(ob)ob.innerHTML='<p class="empty-state">Enter your details above and click the button.</p>';if(inText)inText.focus();});
if(copyBtn)copyBtn.addEventListener('click',function(){
var val=outText&&outText.value!==undefined?outText.value:outText.textContent;
if(!val)return;copyText(val,this,'Copied!','Copy Result');
});
if(downloadBtn)downloadBtn.addEventListener('click',function(){
var val=outText&&outText.value!==undefined?outText.value:outText.textContent;
if(!val)return;
var blob=new Blob([val],{type:'text/plain'});var a=document.createElement('a');a.href=URL.createObjectURL(blob);a.download='website-ranking-checker.txt';document.body.appendChild(a);a.click();document.body.removeChild(a);URL.revokeObjectURL(a.href);
});
})();
</script>
