(function(){
var inText=document.getElementById('inText'),outText=document.getElementById('outText');
var runBtn=document.getElementById('runBtn'),clearBtn=document.getElementById('clearBtn'),copyBtn=document.getElementById('copyBtn'),downloadBtn=document.getElementById('downloadBtn');

function copyText(text,btn,done,orig){
 function d(){btn.textContent=done;setTimeout(function(){btn.textContent=orig;},2000);}
 if(navigator.clipboard&&navigator.clipboard.writeText){navigator.clipboard.writeText(text).then(d);}
 else{var ta=document.createElement('textarea');ta.value=text;ta.style.position='fixed';ta.style.opacity='0';document.body.appendChild(ta);ta.select();document.execCommand('copy');document.body.removeChild(ta);d();}
}


function run(){
 var topic=inText.value.trim();if(!topic)throw new Error('Please enter a topic.');
 var words=topic.toLowerCase().replace(/[^a-z0-9 ]/g,'').split(/\s+/).filter(Boolean);
 var tags=[];
 var base=words.join('');
 if(words.length>1&&tags.indexOf('#'+base)<0)tags.push('#'+base);
 words.forEach(function(w){if(tags.indexOf('#'+w)<0&&w.length>2)tags.push('#'+w);});
 if(words.length>=2){for(var i=0;i<words.length-1;i++){var c=words[i]+words[i+1];if(tags.indexOf('#'+c)<0)tags.push('#'+c);}}
 ['youtube','video','tutorial','howto','guide','tips','viral','shorts','fyp','trending','explore','daily','new','content','creator'].forEach(function(g){if(tags.indexOf('#'+g)<0)tags.push('#'+g);});
 tags=tags.slice(0,15);
 outText.value=tags.join(' ');
 var ob=document.getElementById('outBox');ob.style.display='block';ob.innerHTML='<p class="muted">'+tags.length+' hashtags (YouTube supports up to 15).</p>';
}
function runSafe(){try{run();}catch(e){if(outText)outText.value='Error: '+e.message;else if(document.getElementById('outBox'))document.getElementById('outBox').innerHTML='<p class="muted">Error: '+String(e.message).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;')+'</p>';}}
if(runBtn)runBtn.addEventListener('click',runSafe);
if(clearBtn)clearBtn.addEventListener('click',function(){if(inText)inText.value='';if(outText)outText.value='';var ob=document.getElementById('outBox');if(ob)ob.innerHTML='<p class="empty-state">Enter your details above and click the button.</p>';if(inText)inText.focus();});
if(copyBtn)copyBtn.addEventListener('click',function(){
var val=outText&&outText.value!==undefined?outText.value:outText.textContent;
if(!val)return;copyText(val,this,'Copied!','Copy Result');
});
if(downloadBtn)downloadBtn.addEventListener('click',function(){
var val=outText&&outText.value!==undefined?outText.value:outText.textContent;
if(!val)return;
var blob=new Blob([val],{type:'text/plain'});var a=document.createElement('a');a.href=URL.createObjectURL(blob);a.download='youtube-hashtag-generator.txt';document.body.appendChild(a);a.click();document.body.removeChild(a);URL.revokeObjectURL(a.href);
});
})();
