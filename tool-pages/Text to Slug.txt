<script>
(function(){

var textInput=document.getElementById('textInput'),slugOutput=document.getElementById('slugOutput');
var clearBtn=document.getElementById('clearBtn'),copyBtn=document.getElementById('copyBtn');
var separator=document.getElementById('separator');
var optLower=document.getElementById('optLower'),optTrim=document.getElementById('optTrim'),optRemoveSpecial=document.getElementById('optRemoveSpecial');

function toSlug(){
var text=textInput.value;
if(optTrim.checked)text=text.trim();
if(optRemoveSpecial.checked)text=text.replace(/[^a-zA-Z0-9\s-]/g,'');
var sep=separator.value;
var slug=text.toLowerCase().replace(/\s+/g,sep).replace(new RegExp('\\'+sep+'+','g'),sep).replace(new RegExp('^\\'+sep+'+|\\'+sep+'+$','g'),'');
if(!optLower.checked)slug=text.replace(/\s+/g,sep).replace(new RegExp('\\'+sep+'+','g'),sep).replace(new RegExp('^\\'+sep+'+|\\'+sep+'+$','g'),'');
slugOutput.textContent=slug;
}

textInput.addEventListener('input',toSlug);
separator.addEventListener('change',toSlug);
optLower.addEventListener('change',toSlug);
optTrim.addEventListener('change',toSlug);
optRemoveSpecial.addEventListener('change',toSlug);

clearBtn.addEventListener('click',function(){textInput.value='';slugOutput.textContent='';textInput.focus()});
copyBtn.addEventListener('click',function(){if(slugOutput.textContent){var done=function(){copyBtn.innerHTML='<svg fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M5 13l4 4L19 7"/></svg> Copied!';setTimeout(function(){copyBtn.innerHTML='<svg fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M8 16H6a2 2 0 01-2-2V6a2 2 0 012-2h8a2 2 0 012 2v2m-6 12h8a2 2 0 002-2v-8a2 2 0 00-2-2h-8a2 2 0 00-2 2v8a2 2 0 002 2z"/></svg> Copy'},2000)};if(navigator.clipboard&&navigator.clipboard.writeText){navigator.clipboard.writeText(slugOutput.textContent).then(done)}else{var ta=document.createElement('textarea');ta.value=slugOutput.textContent;ta.style.position='fixed';ta.style.opacity='0';document.body.appendChild(ta);ta.select();document.execCommand('copy');document.body.removeChild(ta);done()}}});
})();
</script>
