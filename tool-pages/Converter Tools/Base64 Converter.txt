(function(){

var inputText=document.getElementById('inputText'),outputText=document.getElementById('outputText');
function toBase64(t){return btoa(unescape(encodeURIComponent(t)))}
function fromBase64(t){try{var b=atob(t.replace(/\s/g,''));var bytes=new Uint8Array(b.length);for(var i=0;i<b.length;i++)bytes[i]=b.charCodeAt(i);return new TextDecoder().decode(bytes)}catch(e){throw e}}
function copyTo(btn,el){if(!el.value)return;var h=btn.innerHTML;
var done=function(){btn.innerHTML='<svg fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M5 13l4 4L19 7"/></svg> Copied!';setTimeout(function(){btn.innerHTML=h},2000)};
if(navigator.clipboard&&navigator.clipboard.writeText){navigator.clipboard.writeText(el.value).then(done);}
else{var ta=document.createElement('textarea');ta.value=el.value;ta.style.position='fixed';ta.style.opacity='0';document.body.appendChild(ta);ta.select();try{document.execCommand('copy');done();}catch(e){}document.body.removeChild(ta);}}

document.getElementById('encodeBtn').addEventListener('click',function(){if(!inputText.value){alert('Please enter text or Base64 to process.');return;}outputText.value=toBase64(inputText.value);outputText.className='tool-textarea'});
document.getElementById('decodeBtn').addEventListener('click',function(){if(!inputText.value){alert('Please enter text or Base64 to process.');return;}try{outputText.value=fromBase64(inputText.value);outputText.className='tool-textarea'}catch(e){outputText.value='Error: Invalid Base64 string';outputText.className='tool-textarea'}});
document.getElementById('clearInp').addEventListener('click',function(){inputText.value='';outputText.value='';inputText.focus()});
document.getElementById('copyInpBtn').addEventListener('click',function(){copyTo(this,inputText)});
document.getElementById('copyOutBtn').addEventListener('click',function(){copyTo(this,outputText)});


})();
