<script>
var STATUS_MAP={200:'OK',201:'Created',204:'No Content',301:'Moved Permanently',302:'Found',304:'Not Modified',400:'Bad Request',401:'Unauthorized',403:'Forbidden',404:'Not Found',429:'Too Many Requests',500:'Internal Server Error',502:'Bad Gateway',503:'Service Unavailable'};
function getBadgeClass(c){if(c>=200&&c<300)return'ok';if(c>=300&&c<400)return'redirect';if(c>=400&&c<500)return'error';return'server'}
function renderResult(url,status,meaning){
var d=document.createElement('div');d.style.cssText='display:flex;align-items:center;gap:12px;padding:10px 14px;border:2px solid var(--border);border-radius:8px;background:var(--bg);flex-wrap:wrap';
var bc=getBadgeClass(status),bg=bc==='ok'?'#D1FAE5':bc==='redirect'?'#FEF3C7':bc==='error'?'#FEE2E2':'#FDE68A',tc=bc==='ok'?'#065F46':bc==='redirect'?'#92400E':bc==='error'?'#991B1B':'#78350F';
var safeUrl=url.replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');
var safeMeaning=meaning.replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;');
d.innerHTML='<span style="padding:3px 10px;border-radius:6px;font-size:.8rem;font-weight:600;white-space:nowrap;background:'+bg+';color:'+tc+'">'+status+'</span><span style="font-size:.85rem;word-break:break-all;flex:1;min-width:120px">'+safeUrl+'</span><span style="font-size:.8rem;color:var(--text-secondary)">'+safeMeaning+'</span>';
return d;
}
async function checkUrl(url){
try{
var proxy='https://api.allorigins.win/get?url='+encodeURIComponent(url);
var r=await fetch(proxy),j=await r.json();
if(j&&j.status&&j.status.http_code){return{url:url,status:j.status.http_code,meaning:STATUS_MAP[j.status.http_code]||'Status code'};}
return{url:url,status:0,meaning:'No status returned'};}
catch(e){return{url:url,status:0,meaning:'Failed to connect'}}}
async function checkSingle(){
var input=document.getElementById('urlInput'),url=input.value.trim();
if(!url)return;if(!url.startsWith('http'))url='https://'+url;
var res=document.getElementById('results');res.innerHTML='<p style="color:var(--text-secondary)">Checking...</p>';
var r=await checkUrl(url);res.innerHTML='';res.appendChild(renderResult(r.url,r.status,r.meaning));}
async function checkBulk(){
var lines=document.getElementById('bulkInput').value.split('\n').map(function(l){return l.trim()}).filter(Boolean);
if(!lines.length)return;var res=document.getElementById('results');
res.innerHTML='<p style="color:var(--text-secondary)">Checking '+lines.length+' URLs...</p>';
var results=[];for(var i=0;i<lines.length;i++){var url=lines[i];if(!url.startsWith('http'))url='https://'+url;results.push(await checkUrl(url));}
res.innerHTML='';results.forEach(function(r){res.appendChild(renderResult(r.url,r.status,r.meaning))});}
document.getElementById('urlInput').addEventListener('keydown',function(e){if(e.key==='Enter')checkSingle()});
</script>
