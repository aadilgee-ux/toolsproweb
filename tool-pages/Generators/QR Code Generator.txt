<script>
(function(){

var qrInput=document.getElementById('qrInput');
var qrSize=document.getElementById('qrSize');
var qrColor=document.getElementById('qrColor');
var qrColorText=document.getElementById('qrColorText');
var generateBtn=document.getElementById('generateBtn');
var downloadBtn=document.getElementById('downloadBtn');
var qrOutput=document.getElementById('qrOutput');

qrColor.addEventListener('input',function(){qrColorText.value=qrColor.value});
qrColorText.addEventListener('input',function(){if(/^#[0-9A-F]{6}$/i.test(qrColorText.value))qrColor.value=qrColorText.value});

function generateQR(){
var text=qrInput.value.trim();
if(!text){alert('Please enter text or URL');return}
var size=+qrSize.value;
var fgColor=qrColor.value||'#000000';
var bgColor='#ffffff';
qrOutput.innerHTML='<div class="qr-code-container" id="qrCodeContainer"></div><div class="qr-download-info">Size: '+size+'x'+size+' — Color: '+fgColor+'</div>';
downloadBtn.style.display='inline-flex';
new QRCode(document.getElementById('qrCodeContainer'),{text:text,width:size,height:size,colorDark:fgColor,colorLight:bgColor,correctLevel:QRCode.CorrectLevel.M});
}

generateBtn.addEventListener('click',generateQR);
qrInput.addEventListener('keypress',function(e){if(e.key==='Enter')generateQR()});

downloadBtn.addEventListener('click',function(){
var holder=document.getElementById('qrCodeContainer');
if(!holder)return;
var cv=holder.querySelector('canvas');
var im=holder.querySelector('img');
var url=cv?cv.toDataURL('image/png'):(im?im.src:null);
if(!url)return;
var a=document.createElement('a');
a.href=url;
a.download='qrcode-'+Date.now()+'.png';
a.click();
});


})();
</script>
