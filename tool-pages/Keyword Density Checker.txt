<script>
(function(){
  

  var stopWords = new Set(['the','a','an','and','or','but','in','on','at','to','for','of','with','by','from','is','it','its','as','was','are','were','be','been','being','have','has','had','do','does','did','will','would','shall','should','may','might','can','could','this','that','these','those','i','me','my','we','our','you','your','he','him','his','she','her','they','them','their','what','which','who','whom','when','where','why','how','all','each','every','both','few','more','most','other','some','such','no','not','only','own','same','so','than','too','very','just','because','if','then','else','while','about','up','out','off','over','under','again','further','once','here','there','also','after','before','above','below','between']);

  document.getElementById('analyzeBtn').addEventListener('click', function(){
    var text = document.getElementById('inputText').value.trim();
    var minLen = parseInt(document.getElementById('minLength').value);
    var topCount = parseInt(document.getElementById('topCount').value);

    if(!text){
      alert('Please enter some text to analyze.');
      return;
    }

    var words = text.toLowerCase().replace(/[^a-z0-9\s]/g, '').split(/\s+/).filter(function(w){ return w.length >= minLen && !stopWords.has(w); });
    var totalWords = text.toLowerCase().replace(/[^a-z0-9\s]/g, '').split(/\s+/).filter(function(w){ return w.length > 0; }).length;
    var uniqueWords = new Set(words).size;

    var freq = {};
    words.forEach(function(w){ freq[w] = (freq[w] || 0) + 1; });

    var sorted = Object.keys(freq).sort(function(a,b){ return freq[b] - freq[a]; }).slice(0, topCount);
    var maxFreq = freq[sorted[0]] || 1;

    // Stats
    document.getElementById('statsDisplay').innerHTML =
      '<div class="stats-grid">' +
        '<div class="stat-box"><div class="stat-num">' + totalWords + '</div><div class="stat-label">Total Words</div></div>' +
        '<div class="stat-box"><div class="stat-num">' + uniqueWords + '</div><div class="stat-label">Unique Words</div></div>' +
        '<div class="stat-box"><div class="stat-num">' + sorted.length + '</div><div class="stat-label">Keywords Found</div></div>' +
      '</div>';

    // Table
    var html = '<table class="kw-table"><thead><tr><th>Keyword</th><th>Count</th><th>Density</th><th></th></tr></thead><tbody>';
    sorted.forEach(function(word){
      var count = freq[word];
      var density = ((count / totalWords) * 100).toFixed(2);
      var barWidth = ((count / maxFreq) * 100).toFixed(0);
      html += '<tr><td>' + word + '</td><td>' + count + '</td><td>' + density + '%</td><td><div class="kw-bar"><div class="kw-bar-fill" style="width:' + barWidth + '%"></div></div></td></tr>';
    });
    html += '</tbody></table>';
    document.getElementById('kwTable').innerHTML = html;

    // Text output
    var output = '=== KEYWORD DENSITY REPORT ===\n\n';
    output += 'Total Words: ' + totalWords + '\n';
    output += 'Unique Words: ' + uniqueWords + '\n\n';
    output += 'TOP KEYWORDS:\n';
    output += '-----------------------------\n';
    sorted.forEach(function(word, i){
      var count = freq[word];
      var density = ((count / totalWords) * 100).toFixed(2);
      output += (i+1) + '. ' + word + ' — ' + count + ' times (' + density + '%)\n';
    });
    document.getElementById('textOutput').textContent = output;

    // Badges
    var badges = '';
    if(totalWords >= 300){ badges += '<div class="output-badge"><svg class="badge-good" fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M9 12l2 2 4-4"/></svg>Good Word Count</div>'; }
    else { badges += '<div class="output-badge"><svg class="badge-warn" fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M12 9v2m0 4h.01"/></svg>Short Text</div>'; }
    if(sorted.length > 0){
      var topDensity = ((freq[sorted[0]] / totalWords) * 100).toFixed(2);
      if(topDensity <= 3){ badges += '<div class="output-badge"><svg class="badge-good" fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M9 12l2 2 4-4"/></svg>Balanced Density</div>'; }
      else { badges += '<div class="output-badge"><svg class="badge-warn" fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M12 9v2m0 4h.01"/></svg>High Keyword Density</div>'; }
    }
    document.getElementById('badgesContainer').innerHTML = badges;
  });

  document.getElementById('copyBtn').addEventListener('click', function(){
    var text = document.getElementById('textOutput').textContent;
    if(text){
      var done = function(){ document.getElementById('copyBtn').innerHTML = '<svg fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M9 12l2 2 4-4"/></svg> Copied!'; setTimeout(function(){ document.getElementById('copyBtn').innerHTML = '<svg fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M8 16H6a2 2 0 01-2-2V6a2 2 0 012-2h8a2 2 0 012 2v2m-6 12h8a2 2 0 002-2v-8a2 2 0 00-2-2h-8a2 2 0 00-2 2v8a2 2 0 002 2z"/></svg> Copy'; }, 2000); };
      if(navigator.clipboard && navigator.clipboard.writeText){
        navigator.clipboard.writeText(text).then(done);
      } else {
        var ta = document.createElement('textarea');
        ta.value = text;
        ta.style.position = 'fixed';
        ta.style.opacity = '0';
        document.body.appendChild(ta);
        ta.select();
        document.execCommand('copy');
        document.body.removeChild(ta);
        done();
      }
    }
  });
})();
</script>
