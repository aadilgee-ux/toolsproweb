<script>
(function(){
  

  var invDate=document.getElementById('invDate');invDate.value=new Date().toISOString().split('T')[0];
  var currencySymbols={USD:'$',EUR:'\u20AC',GBP:'\u00A3',INR:'\u20B9',AUD:'A$',CAD:'C$'};
  var items=[];var itemCounter=0;

  function getSymbol(){return currencySymbols[document.getElementById('currency').value]||'$';}
  function fmt(n){return getSymbol()+parseFloat(n||0).toFixed(2);}

  function recalc(){
    var sub=0;items.forEach(function(it){sub+=it.qty*it.price;});
    var tax=sub*(parseFloat(document.getElementById('taxRate').value)/100);
    var disc=sub*(parseFloat(document.getElementById('discRate').value)/100);
    var total=sub+tax-disc;
    document.getElementById('subtotal').textContent=fmt(sub);
    document.getElementById('taxAmt').textContent=fmt(tax);
    document.getElementById('discAmt').textContent='-'+fmt(disc);
    document.getElementById('totalAmt').textContent=fmt(total);
  }

  function esc(s){return String(s||'').replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;').replace(/'/g,'&#39;');}

  function renderItems(){
    var list=document.getElementById('itemsList');list.innerHTML='';
    items.forEach(function(it,idx){
      var div=document.createElement('div');div.style.cssText='display:grid;grid-template-columns:2fr 1fr 1fr 1fr auto;gap:8px;align-items:end;margin-bottom:10px;';
      div.innerHTML='<div class="form-group" style="margin:0"><input class="form-input item-desc" data-idx="'+idx+'" value="'+esc(it.desc)+'" placeholder="Item description"/></div><div class="form-group" style="margin:0"><input class="form-input item-qty" data-idx="'+idx+'" type="number" min="1" value="'+esc(it.qty)+'"/></div><div class="form-group" style="margin:0"><input class="form-input item-price" data-idx="'+idx+'" type="number" min="0" step="0.01" value="'+esc(it.price)+'"/></div><div class="form-group" style="margin:0"><input class="form-input item-total" value="'+esc(fmt(it.qty*it.price))+'" readonly style="background:var(--bg);font-weight:600"/></div><div class="form-group" style="margin:0"><button class="remove-btn item-remove" data-idx="'+idx+'">&#10005;</button></div>';
      list.appendChild(div);
    });
    document.querySelectorAll('.item-desc').forEach(function(el){el.addEventListener('input',function(){items[+el.dataset.idx].desc=el.value;});});
    document.querySelectorAll('.item-qty').forEach(function(el){el.addEventListener('input',function(){items[+el.dataset.idx].qty=parseFloat(el.value)||1;recalc();renderItems();});});
    document.querySelectorAll('.item-price').forEach(function(el){el.addEventListener('input',function(){items[+el.dataset.idx].price=parseFloat(el.value)||0;recalc();renderItems();});});
    document.querySelectorAll('.item-remove').forEach(function(el){el.addEventListener('click',function(){items.splice(+el.dataset.idx,1);renderItems();recalc();});});
    recalc();
  }

  document.getElementById('addItemBtn').addEventListener('click',function(){items.push({desc:'',qty:1,price:0});renderItems();});
  document.getElementById('taxRate').addEventListener('input',recalc);
  document.getElementById('discRate').addEventListener('input',recalc);
  document.getElementById('currency').addEventListener('change',function(){recalc();renderItems();});

  items=[{desc:'Service / Product',qty:1,price:0}];renderItems();

  document.getElementById('generateBtn').addEventListener('click',function(){
    var jsPDF=window.jspdf?window.jspdf.jsPDF:null;if(!jsPDF){alert('PDF library loading, please try again.');return;}
    var doc=new jsPDF();
    var sym=getSymbol();
    var pageW=doc.internal.pageSize.getWidth();
    var sub=0;items.forEach(function(it){sub+=it.qty*it.price;});
    var tax=sub*(parseFloat(document.getElementById('taxRate').value)/100);
    var disc=sub*(parseFloat(document.getElementById('discRate').value)/100);
    var total=sub+tax-disc;

    doc.setFillColor(99,102,241);doc.rect(0,0,pageW,38,'F');
    doc.setTextColor(255,255,255);doc.setFontSize(20);doc.setFont(undefined,'bold');doc.text('INVOICE',20,18);
    doc.setFontSize(10);doc.setFont(undefined,'normal');doc.text(document.getElementById('invNo').value,20,26);
    doc.text('Date: '+document.getElementById('invDate').value,20,32);
    doc.text('Currency: '+document.getElementById('currency').value,20,36);

    doc.setTextColor(15,23,42);
    var y=50;
    doc.setFont(undefined,'bold');doc.text('From',20,y);y+=6;doc.setFont(undefined,'normal');
    var sn=document.getElementById('senderName').value||'Your Name';doc.text(sn,20,y);y+=5;
    var se=document.getElementById('senderEmail').value;if(se){doc.text(se,20,y);y+=5;}
    var sa=document.getElementById('senderAddr').value;if(sa){var saLines=sa.split('\n');saLines.forEach(function(l){doc.text(l,20,y);y+=5;});}
    y+=4;

    doc.setFont(undefined,'bold');doc.text('Bill To',110,y-8);doc.setFont(undefined,'normal');
    var cn=document.getElementById('clientName').value||'Client Name';doc.text(cn,110,y);y+=5;
    var ce=document.getElementById('clientEmail').value;if(ce){doc.text(ce,110,y);y+=5;}
    var ca=document.getElementById('clientAddr').value;if(ca){var caLines=ca.split('\n');caLines.forEach(function(l){doc.text(l,110,y);y+=5;});}
    y=Math.max(y,60)+6;

    doc.setDrawColor(226,232,240);doc.line(20,y,pageW-20,y);y+=6;
    doc.setFont(undefined,'bold');doc.setFontSize(9);
    doc.text('DESCRIPTION',20,y);doc.text('QTY',120,y);doc.text('PRICE',145,y);doc.text('TOTAL',170,y);
    y+=7;doc.setFont(undefined,'normal');
    items.forEach(function(it){
      var line=it.desc||'-';if(line.length>35)line=line.substring(0,35)+'...';
      doc.text(line,20,y);doc.text(''+it.qty,120,y);doc.text(fmt(it.price),145,y);doc.text(fmt(it.qty*it.price),170,y);
      y+=7;
    });
    doc.line(20,y,pageW-20,y);y+=6;
    doc.setFont(undefined,'bold');
    doc.text('Subtotal:',135,y);doc.text(fmt(sub),170,y);y+=7;
    doc.setFont(undefined,'normal');
    doc.text('Tax ('+document.getElementById('taxRate').value+'%):',135,y);doc.text(fmt(tax),170,y);y+=7;
    doc.text('Discount ('+document.getElementById('discRate').value+'%):',135,y);doc.text('-'+fmt(disc),170,y);y+=7;
    doc.setFont(undefined,'bold');doc.setFontSize(12);doc.text('Total:',135,y);doc.text(fmt(total),170,y);

    var notes=document.getElementById('invNotes').value;
    if(notes){y+=14;doc.setFont(undefined,'normal');doc.setFontSize(9);doc.text('Notes: '+notes,20,y);}

    doc.save((document.getElementById('invNo').value||'invoice')+'.pdf');
  });
})();
</script>
