<script>
(function(){
  

  document.getElementById('analyzeBtn').addEventListener('click', function(){
    var title = document.getElementById('inputTitle').value.trim();
    var desc = document.getElementById('inputDesc').value.trim();
    var keywords = document.getElementById('inputKeywords').value.trim();
    var ogTitle = document.getElementById('inputOGTitle').value.trim();
    var ogDesc = document.getElementById('inputOGDesc').value.trim();

    if(!title && !desc && !keywords){
      alert('Please enter at least one field to analyze.');
      return;
    }

    var score = 0;
    var maxScore = 100;
    var issues = [];
    var good = [];
    var output = '';

    // Title analysis (30 points)
    if(title){
      if(title.length >= 30 && title.length <= 60){
        score += 30;
        good.push('Title length is optimal (' + title.length + ' chars)');
      } else if(title.length > 0 && title.length < 30){
        score += 15;
        issues.push('Title is too short (' + title.length + ' chars). Aim for 30-60 characters.');
      } else if(title.length > 60){
        score += 10;
        issues.push('Title is too long (' + title.length + ' chars). Keep under 60 characters.');
      }
      output += 'TITLE (' + title.length + ' chars):\n' + title + '\n\n';
    } else {
      issues.push('Missing page title. Title is essential for SEO.');
    }

    // Description analysis (30 points)
    if(desc){
      if(desc.length >= 120 && desc.length <= 160){
        score += 30;
        good.push('Description length is optimal (' + desc.length + ' chars)');
      } else if(desc.length > 0 && desc.length < 120){
        score += 15;
        issues.push('Description is too short (' + desc.length + ' chars). Aim for 120-160 characters.');
      } else if(desc.length > 160){
        score += 10;
        issues.push('Description is too long (' + desc.length + ' chars). Keep under 160 characters.');
      }
      output += 'DESCRIPTION (' + desc.length + ' chars):\n' + desc + '\n\n';
    } else {
      issues.push('Missing meta description. Descriptions improve click-through rates.');
    }

    // Keywords analysis (20 points)
    if(keywords){
      var kwArr = keywords.split(',').map(function(k){return k.trim();}).filter(function(k){return k.length > 0;});
      if(kwArr.length >= 3 && kwArr.length <= 10){
        score += 20;
        good.push('Good number of keywords (' + kwArr.length + ')');
      } else if(kwArr.length > 0 && kwArr.length < 3){
        score += 10;
        issues.push('Too few keywords. Add 3-10 relevant keywords.');
      } else if(kwArr.length > 10){
        score += 5;
        issues.push('Too many keywords (' + kwArr.length + '). Keep 3-10 focused keywords.');
      }
      output += 'KEYWORDS (' + kwArr.length + '):\n' + kwArr.join(', ') + '\n\n';
    }

    // OG Tags analysis (20 points)
    if(ogTitle || ogDesc){
      if(ogTitle){ score += 10; good.push('OG Title present'); }
      else { issues.push('Missing OG Title for social media sharing.'); }
      if(ogDesc){ score += 10; good.push('OG Description present'); }
      else { issues.push('Missing OG Description for social media sharing.'); }
    } else {
      issues.push('No Open Graph tags. Add OG tags for better social media sharing.');
    }

    // Display score
    var scoreColor = score >= 80 ? '#22C55E' : score >= 50 ? '#F59E0B' : '#EF4444';
    var scoreLabel = score >= 80 ? 'Excellent' : score >= 50 ? 'Needs Improvement' : 'Poor';
    document.getElementById('scoreDisplay').innerHTML =
      '<div style="text-align:center;margin-bottom:8px;"><span style="font-size:2rem;font-weight:800;color:'+scoreColor+';">'+score+'</span><span style="font-size:1rem;color:#9B8FB0;">/'+maxScore+'</span></div>' +
      '<div class="score-bar"><div class="score-fill" style="width:'+score+'%;background:'+scoreColor+';"></div></div>' +
      '<div style="text-align:center;font-size:0.85rem;color:'+scoreColor+';font-weight:600;">'+scoreLabel+'</div>';

    // Display output
    var result = '=== ANALYSIS REPORT ===\n\n';
    result += 'SEO Score: ' + score + '/' + maxScore + ' (' + scoreLabel + ')\n\n';
    if(good.length > 0){ result += '--- PASSED ---\n'; good.forEach(function(g){ result += '  + ' + g + '\n'; }); result += '\n'; }
    if(issues.length > 0){ result += '--- ISSUES ---\n'; issues.forEach(function(i){ result += '  - ' + i + '\n'; }); result += '\n'; }
    result += '--- META TAGS ---\n' + output;
    document.getElementById('analysisOutput').textContent = result;

    // Display badges
    var badges = '';
    if(score >= 80){ badges += '<div class="output-badge"><svg class="badge-good" fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M9 12l2 2 4-4m6 2a9 9 0 11-18 0 9 9 0 0118 0z"/></svg>Good SEO Score</div>'; }
    else if(score >= 50){ badges += '<div class="output-badge"><svg class="badge-warn" fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M12 9v2m0 4h.01m-6.938 4h13.856c1.54 0 2.502-1.667 1.732-3L13.732 4c-.77-1.333-2.694-1.333-3.464 0L3.34 16c-.77 1.333.192 3 1.732 3z"/></svg>Needs Improvement</div>'; }
    else { badges += '<div class="output-badge"><svg class="badge-bad" fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M10 14l2-2m0 0l2-2m-2 2l-2-2m2 2l2 2m7-2a9 9 0 11-18 0 9 9 0 0118 0z"/></svg>Poor Score</div>'; }
    if(title){ badges += '<div class="output-badge"><svg class="badge-good" fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M9 12l2 2 4-4"/></svg>Title Present</div>'; }
    if(desc){ badges += '<div class="output-badge"><svg class="badge-good" fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M9 12l2 2 4-4"/></svg>Description Present</div>'; }
    document.getElementById('badgesContainer').innerHTML = badges;
  });

  document.getElementById('copyBtn').addEventListener('click', function(){
    var text = document.getElementById('analysisOutput').textContent;
    if(text){
      var done = function(){ document.getElementById('copyBtn').innerHTML = '<svg fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M9 12l2 2 4-4"/></svg> Copied!'; setTimeout(function(){ document.getElementById('copyBtn').innerHTML = '<svg fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M8 16H6a2 2 0 01-2-2V6a2 2 0 012-2h8a2 2 0 012 2v2m-6 12h8a2 2 0 002-2v-8a2 2 0 00-2-2h-8a2 2 0 00-2 2v8a2 2 0 002 2z"/></svg> Copy'; }, 2000); };
      if(navigator.clipboard && navigator.clipboard.writeText){
        navigator.clipboard.writeText(text).then(done);
      } else {
        var ta = document.createElement('textarea');
        ta.value = text;
        ta.style.position = 'fixed';
        ta.style.opacity = '0';
        document.body.appendChild(ta);
        ta.select();
        document.execCommand('copy');
        document.body.removeChild(ta);
        done();
      }
    }
  });
})();
</script>
