<script>
(function() {

  var generateBtn = document.getElementById('generateBtn');
  var metaOutput = document.getElementById('metaOutput');
  var copyBtn = document.getElementById('copyBtn');

  function escapeHtml(t) { return t.replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;'); }

  generateBtn.addEventListener('click', function() {
    var title = document.getElementById('siteTitle').value.trim();
    var author = document.getElementById('siteAuthor').value.trim();
    var desc = document.getElementById('siteDesc').value.trim();
    var keywords = document.getElementById('siteKeywords').value.trim();
    var indexing = document.getElementById('allowIndex').value;
    var follow = document.getElementById('allowFollow').value;

    var tags = [];
    if (title) tags.push('&lt;title&gt;' + escapeHtml(title) + '&lt;/title&gt;');
    if (desc) tags.push('&lt;meta name="description" content="' + escapeHtml(desc) + '"/&gt;');
    if (keywords) tags.push('&lt;meta name="keywords" content="' + escapeHtml(keywords) + '"/&gt;');
    if (author) tags.push('&lt;meta name="author" content="' + escapeHtml(author) + '"/&gt;');
    tags.push('&lt;meta name="robots" content="' + indexing + ', ' + follow + '"/&gt;');

    metaOutput.innerHTML = tags.join('\n');
  });

  copyBtn.addEventListener('click', function() {
    if (metaOutput.textContent) {
      var text = metaOutput.textContent;
      var done = function() {
        copyBtn.innerHTML = '<svg fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M5 13l4 4L19 7"/></svg> Copied!';
        setTimeout(function() { copyBtn.innerHTML = '<svg fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M8 16H6a2 2 0 01-2-2V6a2 2 0 012-2h8a2 2 0 012 2v2m-6 12h8a2 2 0 002-2v-8a2 2 0 00-2-2h-8a2 2 0 00-2 2v8a2 2 0 002 2z"/></svg> Copy'; }, 2000);
      };
      if (navigator.clipboard && navigator.clipboard.writeText) {
        navigator.clipboard.writeText(text).then(done);
      } else {
        var ta = document.createElement('textarea');
        ta.value = text;
        ta.style.position = 'fixed';
        ta.style.opacity = '0';
        document.body.appendChild(ta);
        ta.select();
        document.execCommand('copy');
        document.body.removeChild(ta);
        done();
      }
    }
  });

  
})();
</script>
