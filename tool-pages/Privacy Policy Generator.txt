<script>
(function(){
  
  document.getElementById('effDate').value=new Date().toISOString().split('T')[0];

  function val(id){return document.getElementById(id).value.trim();}
  function checked(id){return document.getElementById(id).checked;}

  function collectItems(){var items=[];document.querySelectorAll('.checkbox-group input[type=checkbox]').forEach(function(cb){if(cb.checked){var s=cb.nextElementSibling;items.push(s?s.textContent.trim():'');}});return items;}

  function generate(){
    var name=val('siteName')||'Our Website';
    var url=val('siteUrl')||'https://example.com';
    var owner=val('ownerName')||'the Company';
    var email=val('contactEmail')||'privacy@example.com';
    var address=val('bizAddress')||'';
    var date=val('effDate')||new Date().toISOString().split('T')[0];

    var collects=[];
    if(checked('col_name'))collects.push('Name');
    if(checked('col_email'))collects.push('Email address');
    if(checked('col_phone'))collects.push('Phone number');
    if(checked('col_address'))collects.push('Mailing address');
    if(checked('col_payment'))collects.push('Payment information');
    if(checked('col_cookies'))collects.push('Cookies and usage data');
    if(checked('col_analytics'))collects.push('Analytics data');
    if(checked('col_device'))collects.push('Device and browser information');
    if(!collects.length)collects.push('Basic usage data');

    var uses=[];
    if(checked('use_service'))uses.push('to provide and maintain our service');
    if(checked('use_improve'))uses.push('to improve and develop our service');
    if(checked('use_comm'))uses.push('to communicate with you');
    if(checked('use_market'))uses.push('to send you marketing communications');
    if(checked('use_analytics'))uses.push('to analyze usage patterns');
    if(checked('use_legal'))uses.push('to comply with legal obligations');
    if(!uses.length)uses.push('to provide and maintain our service');

    var tpServices=[];
    if(checked('tp_google'))tpServices.push('Google Analytics');
    if(checked('tp_stripe'))tpServices.push('Stripe');
    if(checked('tp_paypal'))tpServices.push('PayPal');
    if(checked('tp_fb'))tpServices.push('Facebook Pixel');
    if(checked('tp_ads'))tpServices.push('Google Ads');
    if(checked('tp_cloudflare'))tpServices.push('Cloudflare');

    var isChild=val('childrenPolicy')==='directed';

    var policy='';
    policy+='Privacy Policy\n\n';
    policy+='Last updated: '+date+'\n\n';
    policy+=owner+' ("us", "we", or "our") operates the '+name+' website ('+url+'). This page informs you of our policies regarding the collection, use, and disclosure of personal information when you use our service.\n\n';
    policy+='By using our service, you agree to the collection and use of information in accordance with this policy.\n\n';

    policy+='1. INFORMATION WE COLLECT\n\n';

    policy+='Personal Data: While using our service, we may ask you to provide us with certain personally identifiable information that can be used to contact or identify you. This may include:\n\n';
    collects.forEach(function(c){policy+='  - '+c+'\n';});
    policy+='\n';

    policy+='2. HOW WE USE YOUR INFORMATION\n\n';
    policy+=name+' uses the collected data for various purposes:\n\n';
    uses.forEach(function(u,i){policy+='  '+(i+1)+'. '+u.charAt(0).toUpperCase()+u.slice(1)+'\n';});
    policy+='\n';

    policy+='3. DATA SECURITY\n\n';

      policy+='4. THIRD-PARTY SERVICES\n\n';

These third parties have access to your personal data only to perform these tasks on our behalf and are obligated not to disclose or use it for any other purpose.\n\n';

    policy+='  - The right to data portability\n';
    policy+='  - The right to object to processing\n\n';

      policy+='7. CHILDREN\'S PRIVACY\n\n';

  document.getElementById('generateBtn').addEventListener('click',function(){

  function buildHtml(){
    var name=val('siteName')||'Our Website';
    var owner=val('ownerName')||'the Company';
    var email=val('contactEmail')||'privacy@example.com';
    var date=val('effDate')||new Date().toISOString().split('T')[0];
    var text=document.getElementById('preview').textContent;
    var lines=text.split('\n');
    var html='<!DOCTYPE html>\n<html lang="en">\n<head>\n<meta charset="UTF-8">\n<meta name="viewport" content="width=device-width,initial-scale=1">\n<title>Privacy Policy - '+name+'</title>\n<style>\nbody{font-family:system-ui,-apple-system,sans-serif;max-width:800px;margin:40px auto;padding:0 20px;line-height:1.8;color:#1a1a1a}\nh1{font-size:1.8rem;border-bottom:2px solid #6366F1;padding-bottom:10px}\nh2{font-size:1.2rem;margin-top:24px}\np{margin:12px 0}\nul{margin:12px 0 12px 20px}\nli{margin:6px 0}\n.date{color:#666;margin-bottom:24px}\n</style>\n</head>\n<body>\n';
    html+='<h1>Privacy Policy</h1>\n';
    html+='<p class="date">Last updated: '+date+'</p>\n';
    var sections=text.split(/\n\n/);
    sections.forEach(function(s){
      var l=s.trim();
      if(!l)return;
      if(/^\d+\.\s/.test(l)){
        html+='<h2>'+l+'</h2>\n';
      }else if(l.startsWith('  - ')){
        html+='<ul><li>'+l.replace(/^  - /,'')+'</li></ul>\n';
      }else if(/^\s+\d+\.\s/.test(l)){
        html+='<p>'+l.trim()+'</p>\n';
      }else{
        html+='<p>'+l+'</p>\n';
      }
    });
    html+='</body>\n</html>';
    return html;
  }

  function copyText(text){
    if(navigator.clipboard&&navigator.clipboard.writeText){navigator.clipboard.writeText(text);return;}
    var ta=document.createElement('textarea');ta.value=text;ta.style.position='fixed';ta.style.left='-9999px';
    document.body.appendChild(ta);ta.select();document.execCommand('copy');document.body.removeChild(ta);
  }

  document.getElementById('copyText').addEventListener('click',function(){
    var text=document.getElementById('preview').textContent;
    if(!text){document.getElementById('generateBtn').click();text=document.getElementById('preview').textContent;}
    copyText(text);
    document.getElementById('copyText').textContent='Copied!';setTimeout(function(){document.getElementById('copyText').innerHTML='&#128203; Copy';},1500);
  });

  document.getElementById('copyHtml').addEventListener('click',function(){
    if(!htmlContent)htmlContent=buildHtml();
    copyText(htmlContent);
    document.getElementById('copyHtml').textContent='Copied!';setTimeout(function(){document.getElementById('copyHtml').innerHTML='&#128196; Copy as HTML';},1500);
  });

  document.getElementById('downloadTxt').addEventListener('click',function(){
    var text=document.getElementById('preview').textContent;
    if(!text){document.getElementById('generateBtn').click();text=document.getElementById('preview').textContent;}
    var blob=new Blob([text],{type:'text/plain'});
    var a=document.createElement('a');a.href=URL.createObjectURL(blob);a.download='privacy-policy.txt';a.click();
  });

  document.getElementById('downloadHtml').addEventListener('click',function(){
    if(!htmlContent)htmlContent=buildHtml();
    var blob=new Blob([htmlContent],{type:'text/html'});
    var a=document.createElement('a');a.href=URL.createObjectURL(blob);a.download='privacy-policy.html';a.click();
  });

  document.getElementById('generateBtn').click();
})();
