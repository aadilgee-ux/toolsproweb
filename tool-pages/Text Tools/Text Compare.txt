<script>
(function(){
var inText=document.getElementById('inText'),outText=document.getElementById('outText');
var runBtn=document.getElementById('runBtn'),clearBtn=document.getElementById('clearBtn'),copyBtn=document.getElementById('copyBtn'),downloadBtn=document.getElementById('downloadBtn');
var textA=document.getElementById('textA'),textB=document.getElementById('textB');
var modeCharBtn=document.getElementById('modeChar'),modeWordBtn=document.getElementById('modeWord'),modeLineBtn=document.getElementById('modeLine');

function escHtml(s){return s.replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;');}
function run(){var a=textA.value.replace(/\r\n?/g,'\n'),b=textB.value.replace(/\r\n?/g,'\n');
var la=a.split('\n'),lb=b.split('\n');
var html='';
for(var i=0;i<Math.max(la.length,lb.length);i++){
 var A=i<la.length?la[i]:'',B=i<lb.length?lb[i]:'';
 if(A===B){html+='<div class="diff-line diff-same">'+escHtml(A)+'</div>';}
 else{html+='<div class="diff-line diff-removed">- '+escHtml(A)+'</div><div class="diff-line diff-added">+ '+escHtml(B)+'</div>';}
}
outText.innerHTML=html||'<div class="diff-line diff-same">Texts are identical.</div>';}
function runSafe(){try{run();}catch(e){if(outText)outText.value='Error: '+e.message;}}
if(runBtn)runBtn.addEventListener('click',runSafe);
if(clearBtn)clearBtn.addEventListener('click',function(){if(inText)inText.value='';if(outText)outText.value='';if(inText)inText.focus();});
if(copyBtn)copyBtn.addEventListener('click',function(){
var val=outText&&outText.value!==undefined?outText.value:outText.textContent;
if(!val)return;
var done=function(){copyBtn.innerHTML='Copied!';setTimeout(function(){copyBtn.innerHTML='Copy Result';},2000);};
if(navigator.clipboard&&navigator.clipboard.writeText){navigator.clipboard.writeText(val).then(done)}else{var ta=document.createElement('textarea');ta.value=val;ta.style.position='fixed';ta.style.opacity='0';document.body.appendChild(ta);ta.select();document.execCommand('copy');document.body.removeChild(ta);done()}
});
if(downloadBtn)downloadBtn.addEventListener('click',function(){
var val=outText&&outText.value!==undefined?outText.value:outText.textContent;
if(!val)return;
var blob=new Blob([val],{type:'text/plain'});var a=document.createElement('a');a.href=URL.createObjectURL(blob);a.download='text-compare.txt';document.body.appendChild(a);a.click();document.body.removeChild(a);URL.revokeObjectURL(a.href);
});
})();
</script>
