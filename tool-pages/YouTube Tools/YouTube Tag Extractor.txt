<script>
function extractVideoId(url){
var m=url.match(/(?:v=|youtu\.be\/|embed\/|\/v\/)([a-zA-Z0-9_-]{11})/);
return m?m[1]:null;
}
function esc(t){return t.replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');}

function fetchVideo(){
var url=document.getElementById('ytUrl').value.trim();
var vid=extractVideoId(url);
var preview=document.getElementById('videoPreview');
if(!vid){preview.innerHTML='<p style="color:#EF4444;font-size:0.85rem;margin-top:10px;">Please enter a valid YouTube URL.</p>';return;}
preview.innerHTML='<p style="color:var(--text-secondary);font-size:0.85rem;margin-top:10px;">Loading...</p>';
fetch('https://noembed.com/embed?url=https://www.youtube.com/watch?v='+vid)
.then(function(r){return r.json();})
.then(function(d){
if(d.error){preview.innerHTML='<p style="color:#EF4444;font-size:0.85rem;margin-top:10px;">Could not fetch video info.</p>';return;}
preview.innerHTML='<div style="display:flex;gap:12px;align-items:center;margin-top:12px;padding:10px;border:1px solid var(--border);border-radius:12px;background:var(--surface-2);">'
+'<img src="https://img.youtube.com/vi/'+vid+'/mqdefault.jpg" alt="Thumbnail" style="width:96px;border-radius:8px;display:block;">'
+'<div style="min-width:0;"><h3 style="font-size:0.92rem;color:var(--text);margin-bottom:2px;">'+esc(d.title)+'</h3>'
+'<p style="font-size:0.8rem;color:var(--text-secondary);">'+esc(d.author_name||'')+'</p>'
+'<p style="margin-top:4px;font-size:0.72rem;color:var(--text-secondary);">Video ID: '+vid+'</p></div></div>';
})
.catch(function(){preview.innerHTML='<p style="color:#EF4444;font-size:0.85rem;margin-top:10px;">Error fetching video info. Check the URL.</p>';});
}
document.getElementById('fetchBtn').addEventListener('click',fetchVideo);

function processTags(){
var raw=document.getElementById('tagInput').value.trim();
if(!raw){alert('Please paste some tags first.');return;}
var cleaned=raw.replace(/<[^>]*>/g,' ').replace(/[\n\r]+/g,',');
var tags=cleaned.split(',').map(function(t){return t.trim().toLowerCase().replace(/[^a-z0-9\s\-\+\.#]/g,'').trim();}).filter(function(t){return t.length>0&&t.length<=60;});
var unique=[];var dupes=0;var seen={};
tags.forEach(function(t){if(seen[t]){dupes++;}else{seen[t]=1;unique.push(t);}});
document.getElementById('resultsCard').style.display='block';
document.getElementById('statTotal').textContent=tags.length;
document.getElementById('statUnique').textContent=unique.length;
document.getElementById('statDupes').textContent=dupes;
document.getElementById('tagsDisplay').innerHTML=unique.map(function(t){return '<span style="display:inline-block;background:var(--bg-alt);border:1px solid var(--border);color:var(--text);border-radius:20px;padding:3px 10px;margin:3px;font-size:0.78rem;">'+esc(t)+'</span>';}).join('');
document.getElementById('cleanedTags').value=unique.join(', ');
}
document.getElementById('processBtn').addEventListener('click',processTags);

function clearAll(){
document.getElementById('ytUrl').value='';
document.getElementById('tagInput').value='';
document.getElementById('videoPreview').innerHTML='';
document.getElementById('resultsCard').style.display='none';
}
document.getElementById('clearBtn').addEventListener('click',clearAll);

function downloadTags(content,ext){
var blob=new Blob([content],{type:'text/plain'});
var a=document.createElement('a');a.href=URL.createObjectURL(blob);a.download='youtube-tags.'+ext;a.click();
}
document.getElementById('csvBtn').addEventListener('click',function(){
var tags=document.getElementById('cleanedTags').value;
if(!tags){alert('No tags to export.');return;}
downloadTags('tag\n'+tags.split(', ').join('\n'),'csv');
});
document.getElementById('txtBtn').addEventListener('click',function(){
var tags=document.getElementById('cleanedTags').value;
if(!tags){alert('No tags to export.');return;}
downloadTags(tags.split(', ').join('\n'),'txt');
});
document.getElementById('copyBtn').addEventListener('click',function(){
var tags=document.getElementById('cleanedTags').value;
if(!tags){alert('No tags to copy.');return;}
var btn=this;
function done(){btn.textContent='Copied!';setTimeout(function(){btn.textContent='Copy Tags';},1500);}
if(navigator.clipboard&&navigator.clipboard.writeText){
navigator.clipboard.writeText(tags).then(done);
}else{
var ta=document.createElement('textarea');
ta.value=tags;
ta.style.position='fixed';ta.style.opacity='0';
document.body.appendChild(ta);ta.select();
document.execCommand('copy');
document.body.removeChild(ta);
done();
}
});
