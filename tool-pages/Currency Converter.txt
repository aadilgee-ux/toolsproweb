(function(){

function escHtml(s){return String(s==null?'':s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;');}

 var RATES=null;
 var COMMON=['USD','EUR','GBP','JPY','AUD','CAD','CHF','CNY','INR','PKR','BDT','AED','SAR','TRY','BRL','MXN','RUB','KRW','SGD','NZD','ZAR','NGN','EGP','IDR','MYR','THB','PHP','VND','ARS','CLP','COP'];

 }

 if(n!==0&&(Math.abs(n)<1e-6||Math.abs(n)>=1e15))return n.toExponential(6);
 var r=Math.round(n*1e10)/1e10;

}

 function convert(){

});
})();
