<script>
(function(){
  

  var textInput=document.getElementById('textInput');
  var wordCount=document.getElementById('wordCount');
  var charCount=document.getElementById('charCount');
  var charNoSpace=document.getElementById('charNoSpace');
  var sentenceCount=document.getElementById('sentenceCount');
  var paraCount=document.getElementById('paraCount');
  var readTime=document.getElementById('readTime');

  function count(){
    var text=textInput.value;
    var trimmed=text.trim();
    var words=0;
    var sentences=0;
    if(trimmed!==''){
      words=trimmed.split(/\s+/).filter(function(w){return /[\p{L}\p{N}]/u.test(w);}).length;
      var t=trimmed
        .replace(/(\d)\.(?=\d)/g,'$1')
        .replace(/\b(?:Mr|Mrs|Ms|Dr|Prof|Sr|Jr|St|Rev|Gen|Capt|Col|Corp|Inc|Ltd|Co|etc|e\.?g|i\.?e|vs|No|Ave|Blvd|Dept|Univ|Fig)\./gi,function(m){return m.slice(0,-1);})
        .replace(/\b[A-Z]\.(?:[A-Z]\.)+/g,function(m){return m.replace(/\./g,'');})
        .replace(/\.{3,}/g,'.');
      sentences=t.split(/[.!?]+[)"'\]]*(?=\s|$)/).filter(function(s){return s.trim()!=='';}).length;
    }
    var chars=Array.from(text).length;
    var charsNo=Array.from(text.replace(/\s/g,'')).length;
    var paras=trimmed===''?0:text.split(/\n\s*\n/).filter(function(p){return p.trim().length>0;}).length;
    var readTimeStr;
    if(words===0){readTimeStr='0 min';}
    else if(words<50){readTimeStr=Math.max(1,Math.round(words/200*60))+' sec';}
    else{readTimeStr=Math.ceil(words/200)+' min';}

    wordCount.textContent=words;
    charCount.textContent=chars;
    charNoSpace.textContent=charsNo;
    sentenceCount.textContent=sentences;
    paraCount.textContent=paras;
    readTime.textContent=readTimeStr;
  }

  textInput.addEventListener('input',count);

  document.getElementById('clearBtn').addEventListener('click',function(){
    textInput.value='';
    count();
  });

  count();
})();
</script>
