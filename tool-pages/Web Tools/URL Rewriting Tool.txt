(function(){
var inText=document.getElementById('inText'),outText=document.getElementById('outText');
var runBtn=document.getElementById('runBtn'),clearBtn=document.getElementById('clearBtn'),copyBtn=document.getElementById('copyBtn'),downloadBtn=document.getElementById('downloadBtn');

function copyText(text,btn,done,orig){
 function d(){btn.textContent=done;setTimeout(function(){btn.textContent=orig;},2000);}
 if(navigator.clipboard&&navigator.clipboard.writeText){navigator.clipboard.writeText(text).then(d);}

}
function escHtml(s){return String(s==null?'':s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;');}
function normUrl(u){u=(u||'').trim();if(u&&!/^https?:\/\//i.test(u))u='https://'+u;return u;}


function run(){
 var url=inText.value.trim();if(!url)throw new Error('Please enter a URL (e.g. https://site.com/page.php?id=123).');
 var parsed={};
 try{parsed=new URL(normUrl(url));}catch(e){throw new Error('Invalid URL.');}
 var path=parsed.pathname.split('/').filter(Boolean);
 var file=(path.pop()||'index').replace(/\.[a-z0-9]+$/i,'');
 var slug=slugify(file);
 var params=[];parsed.searchParams.forEach(function(v,k){params.push(k+'='+v);});
 var prettyPath=parsed.origin+'/'+(path.join('/')+'/'+slug).replace(/\/+/g,'\/')+'/';
 var oldPath=parsed.pathname.replace(/^\//,'');
 var escOld=oldPath.split('.').join('[.]');
 var rule='# Redirect the old URL to the new clean URL\nRewriteRule ^'+escOld+'$ '+prettyPath+' [R=301,L]';
 outText.value='Clean URL:\n'+prettyPath+'\n\nRewrite rule (.htaccess):\n'+rule+'\n\nQuery string: '+(params.length?params.join(' & '):'(none)');
 var ob=document.getElementById('outBox');ob.style.display='block';
 ob.innerHTML='<div class="stat-row">'+
  '<div class="stat-card"><span class="stat-label">Old</span><span class="stat-value">'+escHtml(parsed.pathname)+'</span></div>'+
  '<div class="stat-card"><span class="stat-label">New</span><span class="stat-value">'+escHtml(prettyPath)+'</span></div>'+
  '</div>';
}
function slugify(s){return (s||'').toLowerCase().replace(/[^a-z0-9]+/g,'-').replace(/^-+|-+$/g,'')||'page';}
function runSafe(){try{run();}catch(e){if(outText)outText.value='Error: '+e.message;else if(document.getElementById('outBox'))document.getElementById('outBox').innerHTML='<p class="muted">Error: '+String(e.message).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;')+'</p>';}}
if(runBtn)runBtn.addEventListener('click',runSafe);
if(clearBtn)clearBtn.addEventListener('click',function(){if(inText)inText.value='';if(outText)outText.value='';var ob=document.getElementById('outBox');if(ob)ob.innerHTML='<p class="empty-state">Enter your details above and click the button.</p>';if(inText)inText.focus();});

var val=outText&&outText.value!==undefined?outText.value:outText.textContent;
if(!val)return;copyText(val,this,'Copied!','Copy Result');
});

});
