  function generateHash(){
    var text=document.getElementById('inputText').value;
    var algo=document.getElementById('algorithm').value;
    var fmt=document.getElementById('format').value;
    if(!text){alert('Please enter text to hash.');return;}
    var hashes={};
    if(algo==='md5'||algo==='all') hashes.MD5=md5(text);
    if(algo==='sha1'||algo==='all') hashes['SHA-1']=sha1(text);
    if(algo==='sha256'||algo==='all') hashes['SHA-256']=sha256(text);
    var html='';
    var render=function(name,val){
      if(fmt==='upper') val=val.toUpperCase();
      html+='<div class="hash-result"><div class="hash-label">'+name+'</div><div class="hash-value">'+val+'</div></div>';
    };
    if(hashes.MD5) render('MD5',hashes.MD5);
    if(hashes['SHA-1']) render('SHA-1',hashes['SHA-1']);
    if(hashes['SHA-256']) render('SHA-256',hashes['SHA-256']);
    if(algo==='sha512'||algo==='all'){
      if(window.crypto&&window.crypto.subtle){
        var buf=Uint8Array.from(unescape(encodeURIComponent(text)),function(c){return c.charCodeAt(0)});
        crypto.subtle.digest('SHA-512',buf).then(function(digest){
          var arr=Array.from(new Uint8Array(digest));
          var hex=arr.map(function(b){return('0'+b.toString(16)).slice(-2)}).join('');
          render('SHA-512',hex);
          document.getElementById('hashOutput').innerHTML=html;
        }).catch(function(){render('SHA-512','Error: Web Crypto unavailable');document.getElementById('hashOutput').innerHTML=html;});
        return;
      }else{
        render('SHA-512','SHA-512 requires a secure connection (HTTPS)');
      }
    }
    document.getElementById('hashOutput').innerHTML=html;
  }
