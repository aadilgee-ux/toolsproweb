 if(isNaN(c)||isNaN(s))throw new Error('Please enter cost and selling price.');
 if(c<0||s<0)throw new Error('Prices cannot be negative.');
 var profit=s-c;
 var marginPct=s===0?0:profit/s*100;
 var markupPct=c===0?0:profit/c*100;
 outText.value='Profit per unit: $'+profit.toFixed(2)+'\nProfit margin: '+marginPct.toFixed(2)+'%\nMarkup: '+markupPct.toFixed(2)+'%';
 var ob=document.getElementById('outBox');ob.style.display='block';

 '<div class="stat-card"><span class="stat-label">Markup</span><span class="stat-value">'+markupPct.toFixed(1)+'%</span></div>'+
 '<div class="stat-card"><span class="stat-label">Profit</span><span class="stat-value">$'+profit.toFixed(2)+'</span></div>'+
 '</div>';
}

});
