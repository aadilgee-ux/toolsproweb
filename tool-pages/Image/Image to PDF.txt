<script>
(function(){
  

  var files=[];var zone=document.getElementById('uploadZone');var input=document.getElementById('fileInput');var grid=document.getElementById('previewGrid');var btn=document.getElementById('convertBtn');

  zone.addEventListener('click',function(){input.click();});
  zone.addEventListener('dragover',function(e){e.preventDefault();zone.classList.add('dragover');});
  zone.addEventListener('dragleave',function(){zone.classList.remove('dragover');});
  zone.addEventListener('drop',function(e){e.preventDefault();zone.classList.remove('dragover');addFiles(e.dataTransfer.files);});
  input.addEventListener('change',function(){addFiles(input.files);});

  function addFiles(list){
    for(var i=0;i<list.length;i++){
      if(list[i].type.startsWith('image/'))files.push(list[i]);
    }
    renderPreviews();
  }
  function renderPreviews(){
    grid.innerHTML='';
    files.forEach(function(f,idx){
      var div=document.createElement('div');div.className='preview-item';
      var img=document.createElement('img');
      var reader=new FileReader();
      reader.onload=function(e){img.src=e.target.result;};
      reader.readAsDataURL(f);
      var rm=document.createElement('button');rm.className='preview-remove';rm.innerHTML='&times;';
      rm.onclick=function(){files.splice(idx,1);renderPreviews();};
      div.appendChild(img);div.appendChild(rm);grid.appendChild(div);
    });
    btn.disabled=files.length===0;
  }

  btn.addEventListener('click',function(){
    if(files.length===0)return;
    var jsPDF=window.jspdf.jsPDF;
    var pending=files.length;var images=[];
    files.forEach(function(f,idx){
      var reader=new FileReader();
      reader.onload=function(e){
        images[idx]=e.target.result;
        pending--;
        if(pending===0)generatePDF(images);
      };
      reader.readAsDataURL(f);
    });
  });

  function generatePDF(imgs){
    var doc=new jsPDF();
    var loaded=0;
    imgs.forEach(function(src,i){
      var img=new Image();
      img.onload=function(){
        var w=img.width,h=img.height;
        var pw=doc.internal.pageSize.getWidth()-20;
        var ph=doc.internal.pageSize.getHeight()-20;
        var ratio=Math.min(pw/w,ph/h);
        var nw=w*ratio,nh=h*ratio;
        var x=(doc.internal.pageSize.getWidth()-nw)/2;
        var y=(doc.internal.pageSize.getHeight()-nh)/2;
        var canvas=document.createElement('canvas');canvas.width=w;canvas.height=h;
        canvas.getContext('2d').drawImage(img,0,0);
        var fmt='JPEG';
        if(src.indexOf('image/png')===5||src.indexOf('image/webp')===5)fmt='PNG';
        if(i>0)doc.addPage();
        doc.addImage(canvas.toDataURL('image/'+(fmt==='PNG'?'png':'jpeg'),0.92),fmt,x,y,nw,nh);
        loaded++;
        if(loaded===imgs.length)doc.save('images.pdf');
      };
      img.src=src;
    });
  }
})();
</script>
