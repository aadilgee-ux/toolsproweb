<script>
(function(){
  

  var textarea=document.getElementById('inputText');
  textarea.addEventListener('input',function(){document.getElementById('charCount').textContent=textarea.value.length;});

  document.getElementById('convertBtn').addEventListener('click',function(){
    var text=textarea.value;if(!text.trim()){alert('Please enter some text.');return;}
    var fontSize=parseInt(document.getElementById('fontSize').value);
    var fileName=document.getElementById('fileName').value.trim()||'document';
    var jsPDF=window.jspdf.jsPDF;
    var doc=new jsPDF({unit:'pt',format:'a4'});
    var pageW=doc.internal.pageSize.getWidth();
    var pageH=doc.internal.pageSize.getHeight();
    var margin=50;
    var maxW=pageW-margin*2;
    var lines=doc.splitTextToSize(text,maxW);
    var y=margin;
    doc.setFont('courier','normal');
    doc.setFontSize(fontSize);
    lines.forEach(function(line){
      if(y+fontSize>pageH-margin){doc.addPage();y=margin;}
      doc.text(line,margin,y);y+=fontSize*1.5;
    });
    doc.save(fileName+'.pdf');
  });
})();
</script>
