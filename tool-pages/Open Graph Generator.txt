<script>
(function(){
  

  function escapeHtml(t) { return t.replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;'); }

  document.getElementById('generateBtn').addEventListener('click', function(){
    var url = document.getElementById('ogUrl').value.trim();
    var title = document.getElementById('ogTitle').value.trim();
    var desc = document.getElementById('ogDesc').value.trim();
    var image = document.getElementById('ogImage').value.trim();
    var type = document.getElementById('ogType').value;
    var siteName = document.getElementById('ogSiteName').value.trim();

    if(!title && !desc){
      alert('Please enter at least a title or description.');
      return;
    }

    var output = '<!-- Open Graph Meta Tags -->\n';
    output += '<meta property="og:url" content="' + (url || 'https://example.com') + '" />\n';
    output += '<meta property="og:type" content="' + type + '" />\n';
    if(siteName){ output += '<meta property="og:site_name" content="' + siteName + '" />\n'; }
    if(title){ output += '<meta property="og:title" content="' + title + '" />\n'; }
    if(desc){ output += '<meta property="og:description" content="' + desc + '" />\n'; }
    if(image){ output += '<meta property="og:image" content="' + image + '" />\n'; output += '<meta property="og:image:width" content="1200" />\n'; output += '<meta property="og:image:height" content="630" />\n'; }
    output += '\n<!-- Twitter Card Tags -->\n';
    output += '<meta name="twitter:card" content="' + (image ? 'summary_large_image' : 'summary') + '" />\n';
    if(title){ output += '<meta name="twitter:title" content="' + title + '" />\n'; }
    if(desc){ output += '<meta name="twitter:description" content="' + desc + '" />\n'; }
    if(image){ output += '<meta name="twitter:image" content="' + image + '" />\n'; }

    document.getElementById('ogOutput').textContent = output;

    // Preview
    var preview = '<div class="og-preview">';
    preview += '<div class="og-preview-img">' + (image ? '<img src="' + escapeHtml(image) + '" style="width:100%;height:100%;object-fit:cover;" onerror="this.parentElement.innerHTML=\'&#128247;\'"/>' : '&#128247;') + '</div>';
    preview += '<div class="og-preview-body">';
    preview += '<div class="og-preview-domain">' + (url ? escapeHtml(url.replace(/https?:\/\//, '').split('/')[0]) : 'example.com') + '</div>';
    preview += '<div class="og-preview-title">' + (title ? escapeHtml(title) : 'Page Title') + '</div>';
    preview += '<div class="og-preview-desc">' + (desc ? escapeHtml(desc) : 'Page description will appear here when shared on social media.') + '</div>';
    preview += '</div></div>';
    document.getElementById('ogPreview').innerHTML = preview;

    // Badges
    var badges = '';
    if(title){ badges += '<div class="output-badge"><svg class="badge-good" fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M9 12l2 2 4-4"/></svg>Title Set</div>'; }
    if(desc){ badges += '<div class="output-badge"><svg class="badge-good" fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M9 12l2 2 4-4"/></svg>Description Set</div>'; }
    if(image){ badges += '<div class="output-badge"><svg class="badge-good" fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M9 12l2 2 4-4"/></svg>Image Set</div>'; }
    else { badges += '<div class="output-badge"><svg class="badge-warn" fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M12 9v2m0 4h.01"/></svg>No Image</div>'; }
    document.getElementById('badgesContainer').innerHTML = badges;
  });

  document.getElementById('copyBtn').addEventListener('click', function(){
    var text = document.getElementById('ogOutput').textContent;
    if(text){
      var done = function(){ document.getElementById('copyBtn').innerHTML = '<svg fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M9 12l2 2 4-4"/></svg> Copied!'; setTimeout(function(){ document.getElementById('copyBtn').innerHTML = '<svg fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M8 16H6a2 2 0 01-2-2V6a2 2 0 012-2h8a2 2 0 012 2v2m-6 12h8a2 2 0 002-2v-8a2 2 0 00-2-2h-8a2 2 0 00-2 2v8a2 2 0 002 2z"/></svg> Copy'; }, 2000); };
      if(navigator.clipboard && navigator.clipboard.writeText){
        navigator.clipboard.writeText(text).then(done);
      } else {
        var ta = document.createElement('textarea');
        ta.value = text;
        ta.style.position = 'fixed';
        ta.style.opacity = '0';
        document.body.appendChild(ta);
        ta.select();
        document.execCommand('copy');
        document.body.removeChild(ta);
        done();
      }
    }
  });
})();
</script>
