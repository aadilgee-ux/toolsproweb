<script>
(function() {

  var textInput = document.getElementById('textInput');
  var result = document.getElementById('result');
  var clearBtn = document.getElementById('clearBtn');
  var copyBtn = document.getElementById('copyBtn');
  var caseBtns = document.querySelectorAll('.case-btn');
  var currentCase = 'upper';

  function toUpper(t) { return t.toUpperCase(); }
  function toLower(t) { return t.toLowerCase(); }
  function toCapitalize(t) { return t.toLowerCase().replace(/(^|[\s.,!?;:()"\/-])\w/g, function(m) { return m.toUpperCase(); }); }
  function toTitle(t) { return t.replace(/(^|[\s.,!?;:()"\/-])\w/g, function(m) { return m.toUpperCase(); }); }
  function toSentence(t) { return t.toLowerCase().replace(/(^\s*|[.!?]\s+)(\w)/g, function(m, p, c) { return p + c.toUpperCase(); }); }
  function toAlternating(t) { return t.split('').map(function(c, i) { return i % 2 === 0 ? c.toLowerCase() : c.toUpperCase(); }).join(''); }

  function transform() {
    var text = textInput.value;
    var fns = { upper: toUpper, lower: toLower, capitalize: toCapitalize, title: toTitle, sentence: toSentence, alternating: toAlternating };
    result.textContent = text ? fns[currentCase](text) : '';
  }

  caseBtns.forEach(function(btn) {
    btn.addEventListener('click', function() {
      caseBtns.forEach(function(b) { b.classList.remove('active'); });
      btn.classList.add('active');
      currentCase = btn.dataset.case;
      transform();
    });
  });

  textInput.addEventListener('input', transform);
  clearBtn.addEventListener('click', function() { textInput.value = ''; result.textContent = ''; textInput.focus(); });
  copyBtn.addEventListener('click', function() {
    if (result.textContent) {
      var done = function() {
        copyBtn.innerHTML = '<svg fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M5 13l4 4L19 7"/></svg> Copied!';
        setTimeout(function() { copyBtn.innerHTML = '<svg fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M8 16H6a2 2 0 01-2-2V6a2 2 0 012-2h8a2 2 0 012 2v2m-6 12h8a2 2 0 002-2v-8a2 2 0 00-2-2h-8a2 2 0 00-2 2v8a2 2 0 002 2z"/></svg> Copy'; }, 2000);
      };
      if (navigator.clipboard && navigator.clipboard.writeText) {
        navigator.clipboard.writeText(result.textContent).then(done);
      } else {
        var ta = document.createElement('textarea');
        ta.value = result.textContent;
        ta.style.position = 'fixed';
        ta.style.opacity = '0';
        document.body.appendChild(ta);
        ta.select();
        document.execCommand('copy');
        document.body.removeChild(ta);
        done();
      }
    }
  });

  
})();
</script>
